 61    def _normalize_angle(angle: float) -> float:
 62        angle = (angle + math.pi) % (2 * math.pi)
 63        if angle < 0:
 64            angle += 2 * math.pi
 65        return angle - math.pi

 67    def _compute_rotation_segment_to_be_front(
 68        self,
 69        start: OrientedPoint,
 70        target: OrientedPoint,
 71    ) -> RotationSegment:
 72        """Compute a rotation segment so the robot faces the next waypoint.
 73
 74        Args:
 75            start (OrientedPoint): Current robot pose.
 76            target (OrientedPoint): Next waypoint pose (x, y, theta).
 77
 78        Returns:
 79            RotationSegment: Segment rotating in place to face the waypoint.
 80
 81        Raises:
 82            ValueError: If `start.theta` is None.
 83        """
 84        if start.theta is None:
 85            msg = "Start orientation (theta) must be defined."
 86            raise ValueError(msg)
 87
 88        # Compute the absolute heading of the line from start to target
 89        path_theta = start.angle(target)
 90        # If reversing, we want the rear to face the target: add π to the heading
 91        desired_theta = path_theta + (math.pi if self._is_backward else 0)
 92
 93        # Compute minimal rotation from current heading to ``desired_theta``
 94        delta_theta = self._normalize_angle(desired_theta - start.theta)
 95
 96        # Build the intermediate oriented point after rotation
 97        intermediate_pose = OrientedPoint(start.x, start.y, desired_theta)
 98
 99        # Create and return the rotation segment
100        return RotationSegment(
101            start_position=start,
102            end_position=intermediate_pose,
103            duration=self.speed_profiler.angular_speed_profile.get_total_duration(
104                abs(delta_theta),
105            ),
106            rotation=abs(delta_theta),
107            sign=1 if delta_theta > 0 else -1,
108        )

110    def _compute_rotation_segment_to_get_same_orientation(
111        self,
112        start: OrientedPoint,
113        target: OrientedPoint,
114    ) -> RotationSegment:
115        """Compute a rotation segment so final orientation matches ``target``.
116
117        Args:
118            start (OrientedPoint): Robot pose after moving straight to the waypoint.
119            target (OrientedPoint): Desired final waypoint pose (x, y, theta).
120
121        Returns:
122            RotationSegment: Segment rotating in place to align with target orientation.
123
124        Raises:
125            ValueError: If `start.theta` or `target.theta` is None.
126        """
127        if start.theta is None or target.theta is None:
128            msg = "Start and target orientation (theta) must be defined."
129            raise ValueError(msg)
130
131        # Desired final heading: target.theta plus π if reversing
132        desired_theta = target.theta + (math.pi if self._is_backward else 0)
133
134        # Compute the minimal delta angle to rotate from current theta to desired_theta
135        delta_theta = self._normalize_angle(desired_theta - start.theta)
136
137        # Build the intermediate oriented point after rotation
138        intermediate_pose = OrientedPoint(start.x, start.y, desired_theta)
139
140        # Create and return the rotation segment
141        return RotationSegment(
142            start_position=start,
143            end_position=intermediate_pose,
144            duration=self.speed_profiler.angular_speed_profile.get_total_duration(
145                abs(delta_theta),
146            ),
147            rotation=abs(delta_theta),
148            sign=1 if delta_theta > 0 else -1,
149        )

151    def _compute_straight_segment(
152        self,
153        start: OrientedPoint,
154        target: OrientedPoint,
155    ) -> StraightSegment:
156        """Compute straight segment needed to reach the next waypoint.
157
158        Args:
159            start (OrientedPoint): Start pose after initial rotation.
160            target (OrientedPoint): Target waypoint.
161
162        Returns:
163            StraightSegment: Segment that moves in a straight line.
164        """
165        # Compute delta-distance
166        delta_distance = start.distance(
167            target,
168        )  # Use shapely method for more performance
169
170        # Compute intermediate target position (start + distance)
171        target = OrientedPoint(target.x, target.y, start.theta)
172
173        return StraightSegment(
174            start_position=start,
175            end_position=target,
176            duration=self.speed_profiler.linear_speed_profile.get_total_duration(
177                delta_distance,
178            ),
179            distance=delta_distance,
180        )

182    @override
183    def plan_trajectory(self, path: list[OrientedPoint]) -> None:
184        """Build trajectory plan from a list of waypoints.
185
186        Args:
187            path (list[OrientedPoint]): List of oriented points representing the path.
188        """
189        # Initialize the segment list
190        segments: list[BaseSegment] = []
191        # Iterate over each pair of consecutive waypoints
192        for i in range(len(path) - 1):
193            start = path[i]
194            target = path[i + 1]
195
196            # 1. Compute rotation to face the next waypoint
197            rotation_segment = self._compute_rotation_segment_to_be_front(
198                start,
199                target,
200            )
201            segments.append(rotation_segment)
202
203            # 2. Compute straight-line segment to reach the waypoint
204            straight_segment = self._compute_straight_segment(
205                rotation_segment.end_position,
206                target,
207            )
208            segments.append(straight_segment)
209
210            # 3. Compute rotation to align with waypoint orientation
211            # Apply to intermediates if ``respect_intermediate_orientation`` is set
212            # or to the final goal if ``respect_goal_orientation`` is set
213            if target.theta is not None and (
214                self.params.respect_intermediate_orientation
215                or (self.params.respect_goal_orientation and i == len(path) - 2)
216            ):
217                rotation_segment = (
218                    self._compute_rotation_segment_to_get_same_orientation(
219                        straight_segment.end_position,
220                        target,
221                    )
222                )
223                segments.append(rotation_segment)
224            else:
225                # Override orientation to current heading to skip rotation
226                path[i + 1] = OrientedPoint(
227                    path[i + 1].x,
228                    path[i + 1].y,
229                    straight_segment.end_position.theta,
230                )
231
232            # 4. Add a stop segment if a pause is configured
233            if self.params.step_sleep_delay > 0:
234                if target.theta is None:
235                    target = OrientedPoint(
236                        target.x,
237                        target.y,
238                        straight_segment.end_position.theta,
239                    )
240                segments.append(
241                    StopSegment(
242                        start_position=target,
243                        end_position=target,
244                        duration=self.params.step_sleep_delay,
245                    ),
246                )
247
248        # Store the mapped segments for execution
249        self.segments_mapper = SegmentMapper(segments)

252    def get_plan(self) -> TrajectoryPlanCommand:
253        """Retrieve the current motion command based on elapsed time.
254
255        Returns:
256            TrajectoryPlanCommand: The motion command for the current time.
257
258        Raises:
259            RuntimeError: If the trajectory has not been planned yet.
260            ValueError: If required orientation data is missing for segment computation.
261            TypeError: If the segment type is unsupported.
262        """
263        # Get the current time elapsed
264        time_elapsed = self._get_trajectory_time_elapsed()
265
266        # Get the active segment
267        if self.segments_mapper is None:
268            msg = "Trajectory has not been planned yet."
269            raise RuntimeError(msg)
270        segment, local_time = self.segments_mapper.get_segment_at_time(time_elapsed)
271
272        # If no segment is found -> plan is over -> stop the robot at the end path
273        if segment is None:
274            trajectory_plan_command = TrajectoryPlanCommand.create_stop_command(
275                current_position=self.segments_mapper.get_last_segment().end_position,
276            )
277
278        # If the segment is a rotation segment
279        elif isinstance(segment, RotationSegment):
280            if segment.start_position.theta is None:
281                msg = "Segment start position theta must be defined for rotation."
282                raise ValueError(msg)
283
284            th_rotation: float = self.speed_profiler.angular_speed_profile.get_distance(
285                time_elapsed=local_time,
286                distance=segment.rotation,
287            )
288
289            th_theta = segment.start_position.theta + th_rotation * segment.sign
290
291            trajectory_plan_command = TrajectoryPlanCommand(
292                position=OrientedPoint(
293                    segment.start_position.x,
294                    segment.start_position.y,
295                    th_theta,
296                ),
297                linear_speed=0.0,
298                angular_speed=self.speed_profiler.angular_speed_profile.get_speed(
299                    time_elapsed=local_time,
300                    distance=segment.rotation,
301                ),
302            )
303
304        # If the segment is a straight segment
305        elif isinstance(segment, StraightSegment):
306            if segment.start_position.theta is None:
307                msg = (
308                    "Segment start position theta must be defined for straight segment."
309                )
310                raise ValueError(msg)
311
312            th_distance: float = self.speed_profiler.linear_speed_profile.get_distance(
313                time_elapsed=local_time,
314                distance=abs(
315                    segment.distance,
316                ),  # IMPORTANT: Use distance parameter to get the th distance
317            )
318
319            if self._is_backward:
320                th_x = segment.start_position.x - th_distance * math.cos(
321                    segment.start_position.theta,
322                )
323                th_y = segment.start_position.y - th_distance * math.sin(
324                    segment.start_position.theta,
325                )
326            else:
327                th_x = segment.start_position.x + th_distance * math.cos(
328                    segment.start_position.theta,
329                )
330                th_y = segment.start_position.y + th_distance * math.sin(
331                    segment.start_position.theta,
332                )
333
334            trajectory_plan_command = TrajectoryPlanCommand(
335                position=OrientedPoint(th_x, th_y, segment.start_position.theta),
336                linear_speed=self.speed_profiler.linear_speed_profile.get_speed(
337                    time_elapsed=local_time,
338                    distance=abs(
339                        segment.distance,
340                    ),  # IMPORTANT: Use distance parameter to get the th speed
341                ),
342                angular_speed=0.0,
343            )
344
345        # If the segment is a stop segment
346        elif isinstance(segment, StopSegment):
347            trajectory_plan_command = TrajectoryPlanCommand.create_stop_command(
348                current_position=segment.start_position,
349            )
350
351        else:
352            msg = f"Unsupported segment type: {type(segment)}.\n"
353            msg += "Expected RotationSegment, StraightSegment, or StopSegment."
354            raise TypeError(msg)
355
356        # Save the last call time
357        self._last_call_time: float = time_elapsed
358
359        return trajectory_plan_command

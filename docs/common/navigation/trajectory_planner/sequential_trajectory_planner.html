 63    def _normalize_angle(angle: float) -> float:
 64        """Normalize angle to [-pi, pi).
 65
 66        Args:
 67            angle (float): Angle in radians.
 68
 69        Returns:
 70            float: Normalized angle in radians.
 71        """
 72        angle = (angle + math.pi) % (2 * math.pi)
 73        if angle < 0:
 74            angle += 2 * math.pi
 75        return angle - math.pi

 77    def _compute_rotation_segment_to_be_front(
 78        self,
 79        start: OrientedPoint,
 80        target: OrientedPoint,
 81    ) -> RotationSegment:
 82        """Compute a rotation segment so the robot faces the next waypoint.
 83
 84        Args:
 85            start (OrientedPoint): Current robot pose.
 86            target (OrientedPoint): Next waypoint pose (x, y, theta).
 87
 88        Returns:
 89            RotationSegment: Segment rotating in place to face the waypoint.
 90
 91        Raises:
 92            ValueError: If `start.theta` is None.
 93        """
 94        if start.theta is None:
 95            msg = "Start orientation (theta) must be defined."
 96            raise ValueError(msg)
 97
 98        # Compute the absolute heading of the line from start to target
 99        path_theta = start.angle(target)
100        # If reversing, we want the rear to face the target: add π to the heading
101        desired_theta = path_theta + (math.pi if self._is_backward else 0)
102
103        # Compute minimal rotation from current heading to ``desired_theta``
104        delta_theta = self._normalize_angle(desired_theta - start.theta)
105
106        # Build the intermediate oriented point after rotation
107        intermediate_position = OrientedPoint(start.x, start.y, desired_theta)
108
109        # Create and return the rotation segment
110        return RotationSegment(
111            start_position=start,
112            end_position=intermediate_position,
113            duration=self.speed_profiler.angular_speed_profile.get_total_duration(
114                abs(delta_theta),
115            ),
116            rotation=abs(delta_theta),
117            sign=1 if delta_theta > 0 else -1,
118        )

120    def _compute_rotation_segment_to_get_same_orientation(
121        self,
122        start: OrientedPoint,
123        target: OrientedPoint,
124    ) -> RotationSegment:
125        """Compute a rotation segment so final orientation matches ``target``.
126
127        Args:
128            start (OrientedPoint): Robot pose after moving straight to the waypoint.
129            target (OrientedPoint): Desired final waypoint pose (x, y, theta).
130
131        Returns:
132            RotationSegment: Segment rotating in place to align with target orientation.
133
134        Raises:
135            ValueError: If `start.theta` or `target.theta` is None.
136        """
137        if start.theta is None or target.theta is None:
138            msg = "Start and target orientation (theta) must be defined."
139            raise ValueError(msg)
140
141        # Desired final heading: target.theta plus π if reversing
142        desired_theta = target.theta + (math.pi if self._is_backward else 0)
143
144        # Compute the minimal delta angle to rotate from current theta to desired_theta
145        delta_theta = self._normalize_angle(desired_theta - start.theta)
146
147        # Build the intermediate oriented point after rotation
148        intermediate_position = OrientedPoint(start.x, start.y, desired_theta)
149
150        # Create and return the rotation segment
151        return RotationSegment(
152            start_position=start,
153            end_position=intermediate_position,
154            duration=self.speed_profiler.angular_speed_profile.get_total_duration(
155                abs(delta_theta),
156            ),
157            rotation=abs(delta_theta),
158            sign=1 if delta_theta > 0 else -1,
159        )

161    def _compute_straight_segment(
162        self,
163        start: OrientedPoint,
164        target: OrientedPoint,
165    ) -> StraightSegment:
166        """Compute straight segment needed to reach the next waypoint.
167
168        Args:
169            start (OrientedPoint): Start pose after initial rotation.
170            target (OrientedPoint): Target waypoint.
171
172        Returns:
173            StraightSegment: Segment that moves in a straight line.
174        """
175        # Compute delta-distance
176        delta_distance = start.distance(
177            target,
178        )  # Use shapely method for more performance
179
180        # Compute intermediate target position (start + distance)
181        target = OrientedPoint(target.x, target.y, start.theta)
182
183        return StraightSegment(
184            start_position=start,
185            end_position=target,
186            duration=self.speed_profiler.linear_speed_profile.get_total_duration(
187                delta_distance,
188            ),
189            distance=delta_distance,
190        )

192    @override
193    def plan_trajectory(self, path: list[OrientedPoint]) -> None:
194        """Build trajectory plan from a list of waypoints.
195
196        Args:
197            path (list[OrientedPoint]): List of oriented points representing the path.
198        """
199        if path:
200            self._fallback_stop_position = path[-1]
201
202        # Initialize the segment list
203        segments: list[BaseSegment] = []
204        # Iterate over each pair of consecutive waypoints
205        for i in range(len(path) - 1):
206            start = path[i]
207            target = path[i + 1]
208
209            # 1. Compute rotation to face the next waypoint
210            rotation_segment = self._compute_rotation_segment_to_be_front(
211                start,
212                target,
213            )
214            segments.append(rotation_segment)
215
216            # 2. Compute straight-line segment to reach the waypoint
217            straight_segment = self._compute_straight_segment(
218                rotation_segment.end_position,
219                target,
220            )
221            segments.append(straight_segment)
222
223            # 3. Compute rotation to align with waypoint orientation
224            # Apply to intermediates if ``respect_intermediate_orientation`` is set
225            # or to the final goal if ``respect_goal_orientation`` is set
226            if target.theta is not None and (
227                self.params.respect_intermediate_orientation
228                or (self.params.respect_goal_orientation and i == len(path) - 2)
229            ):
230                rotation_segment = (
231                    self._compute_rotation_segment_to_get_same_orientation(
232                        straight_segment.end_position,
233                        target,
234                    )
235                )
236                segments.append(rotation_segment)
237            else:
238                # Override orientation to current heading to skip rotation
239                path[i + 1] = OrientedPoint(
240                    path[i + 1].x,
241                    path[i + 1].y,
242                    straight_segment.end_position.theta,
243                )
244
245            # 4. Add a stop segment if a pause is configured
246            if self.params.step_sleep_delay > 0:
247                if target.theta is None:
248                    target = OrientedPoint(
249                        target.x,
250                        target.y,
251                        straight_segment.end_position.theta,
252                    )
253                segments.append(
254                    StopSegment(
255                        start_position=target,
256                        end_position=target,
257                        duration=self.params.step_sleep_delay,
258                    ),
259                )
260
261        # Store the mapped segments for execution
262        self.segments_mapper = SegmentMapper(segments)
263        self._empty_path_warning_emitted = False
264
265        if not segments:
266            self._logger.warning(
267                "[NAV:Trajectory] Empty trajectory generated; robot will hold position.",
268            )

271    def get_plan(self) -> TrajectoryPlanCommand:
272        """Retrieve the current motion command based on elapsed time.
273
274        Returns:
275            TrajectoryPlanCommand: The motion command for the current time.
276
277        Raises:
278            RuntimeError: If the trajectory has not been planned yet.
279            ValueError: If required orientation data is missing for segment computation.
280            TypeError: If the segment type is unsupported.
281        """
282        # Get the current time elapsed
283        time_elapsed = self._get_trajectory_time_elapsed()
284
285        # Get the active segment
286        if self.segments_mapper is None:
287            msg = "Trajectory has not been planned yet."
288            raise RuntimeError(msg)
289
290        if not self.segments_mapper.segments:
291            if not self._empty_path_warning_emitted:
292                self._logger.warning(
293                    "[NAV:Trajectory] get_plan() called with no segments; "
294                    " returning stop command.",
295                )
296                self._empty_path_warning_emitted = True
297
298            return TrajectoryPlanCommand.create_stop_command(
299                current_position=self._fallback_stop_position,
300            )
301
302        segment, local_time = self.segments_mapper.get_segment_at_time(time_elapsed)
303
304        # If no segment is found -> plan is over -> stop the robot at the end path
305        if segment is None:
306            trajectory_plan_command = TrajectoryPlanCommand.create_stop_command(
307                current_position=self.segments_mapper.get_last_segment().end_position,
308            )
309
310        # If the segment is a rotation segment
311        elif isinstance(segment, RotationSegment):
312            if segment.start_position.theta is None:
313                msg = "Segment start position theta must be defined for rotation."
314                raise ValueError(msg)
315
316            th_rotation: float = self.speed_profiler.angular_speed_profile.get_distance(
317                time_elapsed=local_time,
318                distance=segment.rotation,
319            )
320
321            th_theta = segment.start_position.theta + th_rotation * segment.sign
322
323            trajectory_plan_command = TrajectoryPlanCommand(
324                position=OrientedPoint(
325                    segment.start_position.x,
326                    segment.start_position.y,
327                    th_theta,
328                ),
329                linear_speed=0.0,
330                angular_speed=(
331                    self.speed_profiler.angular_speed_profile.get_speed(
332                        time_elapsed=local_time,
333                        distance=segment.rotation,
334                    )
335                    * segment.sign
336                ),
337            )
338
339        # If the segment is a straight segment
340        elif isinstance(segment, StraightSegment):
341            if segment.start_position.theta is None:
342                msg = (
343                    "Segment start position theta must be defined for straight segment."
344                )
345                raise ValueError(msg)
346
347            th_distance: float = self.speed_profiler.linear_speed_profile.get_distance(
348                time_elapsed=local_time,
349                distance=abs(
350                    segment.distance,
351                ),  # IMPORTANT: Use distance parameter to get the th distance
352            )
353
354            if self._is_backward:
355                th_x = segment.start_position.x - th_distance * math.cos(
356                    segment.start_position.theta,
357                )
358                th_y = segment.start_position.y - th_distance * math.sin(
359                    segment.start_position.theta,
360                )
361            else:
362                th_x = segment.start_position.x + th_distance * math.cos(
363                    segment.start_position.theta,
364                )
365                th_y = segment.start_position.y + th_distance * math.sin(
366                    segment.start_position.theta,
367                )
368
369            trajectory_plan_command = TrajectoryPlanCommand(
370                position=OrientedPoint(th_x, th_y, segment.start_position.theta),
371                linear_speed=(
372                    self.speed_profiler.linear_speed_profile.get_speed(
373                        time_elapsed=local_time,
374                        distance=abs(
375                            segment.distance,
376                        ),  # IMPORTANT: Use distance parameter to get the th speed
377                    )
378                    * (-1 if self._is_backward else 1)
379                ),
380                angular_speed=0.0,
381            )
382
383        # If the segment is a stop segment
384        elif isinstance(segment, StopSegment):
385            trajectory_plan_command = TrajectoryPlanCommand.create_stop_command(
386                current_position=segment.start_position,
387            )
388
389        else:
390            msg = f"Unsupported segment type: {type(segment)}.\n"
391            msg += "Expected RotationSegment, StraightSegment, or StopSegment."
392            raise TypeError(msg)
393
394        # Save the last call time
395        self._last_call_time: float = time_elapsed
396
397        return trajectory_plan_command

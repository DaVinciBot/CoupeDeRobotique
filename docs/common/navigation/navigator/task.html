144        self,
145        current_position: OrientedPoint,

147        """Replan after avoidance without restarting relative delta tasks.

167        last_params.start = current_position

173    def _is_goal_reached(self, current_position: OrientedPoint) -> bool:
174        goal = self._planned_goal or self.params.goal
175        if goal is None:
176            return False
177
178        if current_position.distance(goal) > self.params.position_reached_tolerance_cm:
179            return False
180
181        if current_position.theta is None or goal.theta is None:
182            return True
183
184        return (
185            abs(self._normalize_angle(goal.theta - current_position.theta))
186            <= self.params.angle_reached_tolerance_rad
187        )

189    def _is_finished(self, current_position: OrientedPoint) -> bool:
190        if self._start_time is None or self.state == NavigatorTaskState.FINISHED:
191            return False
192        total_duration = self.trajectory_planner.get_total_duration()
193        elapsed_time = self._get_elapsed_time()
194        if elapsed_time <= total_duration:
195            return False
196        if self._is_goal_reached(current_position):
197            return True
198        finish_delay = self.params.finish_after_expected_end_delay_s
199        return finish_delay is not None and elapsed_time > total_duration + finish_delay

201    def _command_from_current_position(
202        self,
203        planned_cmd: TrajectoryPlanCommand,
204        current_position: OrientedPoint,
205    ) -> TrajectoryPlanCommand:
206        """Rebase the trajectory command on the measured robot pose.
207
208        The trajectory planner produces a time-indexed theoretical pose. The motor
209        controller, however, should chase a local target built from the real
210        odometry so a lagging robot does not keep receiving targets that run away
211        along the ideal trajectory.
212
213        Returns:
214            TrajectoryPlanCommand: Command rebased on the measured pose.
215        """
216        from geometry import OrientedPoint  # noqa: PLC0415
217
218        goal = self._planned_goal or planned_cmd.position
219        trajectory_is_over = (
220            self._get_elapsed_time() >= self.trajectory_planner.get_total_duration()
221        )
222        if trajectory_is_over:
223            return TrajectoryPlanCommand(
224                position=goal,
225                linear_speed=0.0,
226                angular_speed=0.0,
227            )
228
229        if current_position.theta is None:
230            return planned_cmd
231
232        target_theta = current_position.theta
233        has_angular_speed = abs(planned_cmd.angular_speed) > SPEED_EPSILON
234        has_linear_speed = abs(planned_cmd.linear_speed) > SPEED_EPSILON
235
236        if has_angular_speed:
237            target_theta = current_position.theta + (
238                planned_cmd.angular_speed * TRACKING_LOOKAHEAD_S
239            )
240        elif planned_cmd.position.theta is not None:
241            target_theta = planned_cmd.position.theta
242
243        dx = 0.0
244        dy = 0.0
245        if has_linear_speed:
246            heading = (
247                planned_cmd.position.theta
248                if planned_cmd.position.theta is not None
249                else current_position.theta
250            )
251            step = planned_cmd.linear_speed * TRACKING_LOOKAHEAD_S
252            max_step = current_position.distance(goal)
253            if abs(step) > max_step:
254                step = math.copysign(max_step, step)
255            dx = step * math.cos(heading)
256            dy = step * math.sin(heading)
257
258        if not has_linear_speed and not has_angular_speed:
259            target = current_position
260        else:
261            target = OrientedPoint(
262                (current_position.x + dx, current_position.y + dy),
263                target_theta,
264            )
265
266        return TrajectoryPlanCommand(
267            position=target,
268            linear_speed=planned_cmd.linear_speed,
269            angular_speed=planned_cmd.angular_speed,
270        )

272    def handle(
273        self,
274        ally_zone: AllyZone,
275        enemy_zone: EnemyZone,
276    ) -> TrajectoryPlanCommand:
277        """Advance the task and return the next trajectory command.
278
279        Args:
280            ally_zone (AllyZone): The ally zone information.
281            enemy_zone (EnemyZone): The enemy zone information.
282
283        Returns:
284            TrajectoryPlanCommand: Next trajectory command.
285        """
286        if self.state == NavigatorTaskState.NOT_PLANNED:
287            self._plan_task(ally_zone)
288            planned_cmd = self.trajectory_planner.get_plan()
289            self.current_trajectory_command = self._command_from_current_position(
290                planned_cmd,
291                ally_zone.point,
292            )
293            return self.current_trajectory_command
294
295        if self._has_timed_out():
296            return self._abort(ally_zone.point)
297
298        if self.state == NavigatorTaskState.AVOIDING:
299            return self.avoidance.handle(
300                current_navigator_task=self,
301                ally_zone=ally_zone,
302                enemy_zone=enemy_zone,
303            )
304
305        planned_cmd = self.trajectory_planner.get_plan()
306        cmd = self._command_from_current_position(planned_cmd, ally_zone.point)
307
308        if (
309            self._is_finished(ally_zone.point)
310            and self.state != NavigatorTaskState.STABILIZING
311        ):
312            if self.params.stabilization_delay > 0:
313                self._stabilization_start_time = time.time()
314                self.state = NavigatorTaskState.STABILIZING
315            else:
316                self.state = NavigatorTaskState.FINISHED
317
318        if self.state == NavigatorTaskState.STABILIZING:
319            if self._get_stabilization_elapsed() < self.params.stabilization_delay:
320                self.current_trajectory_command = (
321                    TrajectoryPlanCommand.create_stop_command(
322                        current_position=ally_zone.point,
323                    )
324                )
325                return self.current_trajectory_command
326            self.state = NavigatorTaskState.FINISHED
327
328        if self.state == NavigatorTaskState.FINISHED:
329            self.current_trajectory_command = TrajectoryPlanCommand.create_stop_command(
330                current_position=ally_zone.point,
331            )
332            return self.current_trajectory_command
333
334        self.current_trajectory_command = cmd
335
336        avoidance_cmd = self.avoidance.handle(
337            current_navigator_task=self,
338            ally_zone=ally_zone,
339            enemy_zone=enemy_zone,
340        )
341        if self.state == NavigatorTaskState.AVOIDING:
342            return avoidance_cmd
343
344        return self.current_trajectory_command

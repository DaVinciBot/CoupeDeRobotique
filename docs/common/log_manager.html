 18    # Log format: HH:MM:SS.ffffff -> [logger] [file:line] LEVEL | message
 19    LOG_PATTERN = re.compile(
 20        r"(?P<time>\d{2}:\d{2}:\d{2}\.\d+)\s+->\s+"
 21        r"\[(?P<logger>[^\]]+)\]\s+"
 22        r"\[(?P<file>[^:]+):(?P<line>\d+)\]\s+"
 23        r"(?P<level>\w+)\s+\|\s+"
 24        r"(?P<message>.*)",
 25    )

 27    # Category pattern: [CATEGORY] or [CATEGORY:SubCategory]
 28    CATEGORY_PATTERN = re.compile(r"\[([A-Z]+(?::[A-Za-z0-9:]+)?)\]")

 40    def _parse_log_line(
 41        self,
 42        line: str,
 43        offset: int,
 44    ) -> dict[str, Any] | None:
 45        """Parse a single log line.
 46
 47        Args:
 48            line: Log line to parse
 49            offset: Byte offset in file
 50
 51        Returns:
 52            Parsed log entry or None if parse failed
 53        """
 54        match = self.LOG_PATTERN.match(line)
 55        if not match:
 56            return None
 57
 58        data = match.groupdict()
 59
 60        category = None
 61        category_match = self.CATEGORY_PATTERN.search(data["message"])
 62        if category_match:
 63            category = category_match.group(1)
 64
 65        if self.current_date:
 66            timestamp = f"{self.current_date} {data['time']}"
 67        else:
 68            timestamp = data["time"]
 69
 70        return {
 71            "timestamp": timestamp,
 72            "level": data["level"],
 73            "logger_name": data["logger"].strip(),
 74            "file_name": data["file"].strip(),
 75            "line_number": int(data["line"]),
 76            "category": category,
 77            "message": data["message"],
 78            "raw_line": line.strip(),
 79            "line_offset": offset,
 80        }

 82    def _detect_execution_boundary(self, line: str) -> bool:
 83        """Detect if a log line indicates a new execution start.
 84
 85        Args:
 86            line: Log line to check
 87
 88        Returns:
 89            True if this is an execution boundary
 90        """
 91        markers = [
 92            "[BRAIN:Init]",
 93            "Starting brain",
 94            "Team color set to",
 95            "WS_Server.*Initialized",
 96        ]
 97
 98        for marker in markers:
 99            if marker in line:
100                return True
101
102        return False

104    def index_log_file(
105        self,
106        log_file: Path | str,
107        *,
108        force_reindex: bool = False,
109    ) -> int:
110        """Index a log file into the database.
111
112        Args:
113            log_file: Path to log file
114            force_reindex: If True, re-index even if already indexed
115
116        Returns:
117            Number of log entries indexed
118
119        Raises:
120            FileNotFoundError: If the log file does not exist
121        """
122        log_path = Path(log_file)
123        if not log_path.exists():
124            raise FileNotFoundError(f"Log file not found: {log_path}")
125
126        # Extract date from filename (e.g., 2025-11-06.log)
127        self.current_date = log_path.stem
128
129        execution_counter = 1
130        entries_indexed = 0
131        current_offset = 0
132
133        with log_path.open("r", encoding="utf-8", errors="ignore") as f:
134            for line in f:
135                # Track offset for later retrieval
136                line_start = current_offset
137                current_offset += len(line.encode("utf-8"))
138
139                # Detect execution boundaries
140                if self._detect_execution_boundary(line):
141                    execution_counter += 1
142                    self.current_execution_id = (
143                        f"{self.current_date}_exec{execution_counter:03d}"
144                    )
145
146                    # Register execution in database
147                    parsed = self._parse_log_line(line, line_start)
148                    if parsed:
149                        start_time = datetime.fromisoformat(parsed["timestamp"])
150                        self.db.add_execution(
151                            execution_id=self.current_execution_id,
152                            start_time=start_time,
153                            log_file=str(log_path),
154                            description=f"Execution {execution_counter}",
155                        )
156
157                # Ensure we have an execution_id
158                if not self.current_execution_id:
159                    self.current_execution_id = f"{self.current_date}_exec001"
160                    self.db.add_execution(
161                        execution_id=self.current_execution_id,
162                        start_time=datetime.now(),
163                        log_file=str(log_path),
164                        description="Execution 1",
165                    )
166
167                # Parse and index the log line
168                parsed = self._parse_log_line(line, line_start)
169                if parsed:
170                    self.db.add_log_entry(
171                        execution_id=self.current_execution_id,
172                        **parsed,
173                    )
174                    entries_indexed += 1
175
176        return entries_indexed

131        current_offset = 0
132
133        with log_path.open("r", encoding="utf-8", errors="ignore") as f:
134            for line in f:
135                # Track offset for later retrieval
136                line_start = current_offset
137                current_offset += len(line.encode("utf-8"))
138
139                # Detect execution boundaries
140                if self._detect_execution_boundary(line):
141                    execution_counter += 1
142                    self.current_execution_id = (
143                        f"{self.current_date}_exec{execution_counter:03d}"
144                    )
145
146                    # Register execution in database
147                    parsed = self._parse_log_line(line, line_start)
148                    if parsed:
149                        start_time = datetime.fromisoformat(parsed["timestamp"])
150                        self.db.add_execution(
151                            execution_id=self.current_execution_id,
152                            start_time=start_time,
153                            log_file=str(log_path),
154                            description=f"Execution {execution_counter}",
155                        )
156
157                # Ensure we have an execution_id
158                if not self.current_execution_id:
159                    self.current_execution_id = f"{self.current_date}_exec001"
160                    self.db.add_execution(
161                        execution_id=self.current_execution_id,
162                        start_time=datetime.now(),
163                        log_file=str(log_path),
164                        description="Execution 1",
165                    )
166
167                # Parse and index the log line
168                parsed = self._parse_log_line(line, line_start)
169                if parsed:
170                    self.db.add_log_entry(
171                        execution_id=self.current_execution_id,
172                        **parsed,
173                    )
174                    entries_indexed += 1
175
176        return entries_indexed

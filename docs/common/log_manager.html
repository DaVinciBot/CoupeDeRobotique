 27    def _initialize_database(self) -> None:
 28        """Create database and tables if they don't exist."""
 29        self.conn = sqlite3.connect(str(self.db_path))
 30        self.conn.row_factory = sqlite3.Row
 31
 32        cursor = self.conn.cursor()
 33
 34        # Create executions table
 35        cursor.execute("""
 36            CREATE TABLE IF NOT EXISTS executions (
 37                id INTEGER PRIMARY KEY AUTOINCREMENT,
 38                execution_id TEXT UNIQUE NOT NULL,
 39                start_time TEXT NOT NULL,
 40                end_time TEXT,
 41                log_file TEXT NOT NULL,
 42                description TEXT
 43            )
 44        """)
 45
 46        # Create logs table with full indexing
 47        cursor.execute("""
 48            CREATE TABLE IF NOT EXISTS logs (
 49                id INTEGER PRIMARY KEY AUTOINCREMENT,
 50                execution_id TEXT NOT NULL,
 51                timestamp TEXT NOT NULL,
 52                level TEXT NOT NULL,
 53                logger_name TEXT NOT NULL,
 54                file_name TEXT,
 55                line_number INTEGER,
 56                category TEXT,
 57                message TEXT NOT NULL,
 58                raw_line TEXT NOT NULL,
 59                FOREIGN KEY (execution_id) REFERENCES executions(execution_id)
 60            )
 61        """)
 62
 63        # Create indexes for fast searching
 64        cursor.execute(
 65            "CREATE INDEX IF NOT EXISTS idx_execution ON logs(execution_id)",
 66        )
 67        cursor.execute("CREATE INDEX IF NOT EXISTS idx_level ON logs(level)")
 68        cursor.execute(
 69            "CREATE INDEX IF NOT EXISTS idx_logger ON logs(logger_name)",
 70        )
 71        cursor.execute(
 72            "CREATE INDEX IF NOT EXISTS idx_category ON logs(category)",
 73        )
 74        cursor.execute(
 75            "CREATE INDEX IF NOT EXISTS idx_file ON logs(file_name)",
 76        )
 77        cursor.execute(
 78            "CREATE INDEX IF NOT EXISTS idx_timestamp ON logs(timestamp)",
 79        )

 81        self.conn.commit()

 36        Args:
 37            parsed (dict[str, Any]): Currently parsed log entry
 38            last_entry (dict[str, Any] | None): Last indexed log entry from database
 39
 40        Returns:
 41            tuple[bool, bool]: Tuple of (should_skip, should_exit_skip_mode):
 42                - should_skip: True if this entry should be skipped
 43                - should_exit_skip_mode: True if we should exit skip mode after this
 44        """
 45        if not last_entry:
 46            return (False, True)
 47
 48        current_ts = parsed["timestamp"]
 49        last_ts = last_entry["timestamp"]
 50
 51        if current_ts < last_ts:
 52            return (True, False)
 53
 54        if current_ts == last_ts:
 55            current_location = f"{parsed['file_name']}:{parsed['line_number']}"
 56            last_location = f"{last_entry['file_name']}:{last_entry['line_number']}"
 57
 58            if current_location == last_location:
 59                return (True, True)
 60
 61            return (False, False)
 62
 63        return (False, True)
 64

 66        self,
 67        log_file: Path | str,
 68        *,
 69        force_reindex: bool,
 70    ) -> int:
 71        """Index a log file into the database.
 72
 73        Args:
 74            log_file (Path | str): Path to log file
 75            force_reindex (bool):
 76                If True, re-index even if already indexed.
 77
 78        Returns:
 79            int: Number of log entries indexed
 80
 81        Raises:
 82            FileNotFoundError: If the log file does not exist
 83        """
 84        log_path = Path(log_file)
 85        if not log_path.exists():
 86            msg = f"Log file not found: {log_path}"
 87            raise FileNotFoundError(msg)
 88
 89        self.current_date = log_path.stem
 90
 91        last_entry = (
 92            None
 93            if force_reindex
 94            else self.db.get_last_log_entry(
 95                str(log_path),
 96            )
 97        )
 98
 99        existing_executions = self.db.get_executions(str(log_path))
100        existing_exec_ids = {e["execution_id"] for e in existing_executions}
101
102        execution_counter = 0
103        if existing_executions:
104            last_exec_id = existing_executions[-1]["execution_id"]
105            execution_counter = int(last_exec_id.split("_exec")[-1])
106            self.current_execution_id = last_exec_id
107
108        entries_indexed = 0
109        skipping_mode = last_entry is not None
110
111        with log_path.open("r", encoding="utf-8", errors="ignore") as f:
112            for line in f:
113                parsed = parse_log_line(line, current_date=self.current_date)
114                if not parsed:
115                    continue
116
117                if skipping_mode:
118                    should_skip, should_exit_skip_mode = self._should_skip_entry(
119                        parsed,
120                        last_entry,
121                    )
122                    if should_exit_skip_mode:
123                        skipping_mode = False
124                    if should_skip:
125                        continue
126
127                if detect_execution_boundary(line):
128                    new_execution_counter = execution_counter + 1
129                    new_execution_id = (
130                        f"{self.current_date}_exec{new_execution_counter:03d}"
131                    )
132
133                    if new_execution_id not in existing_exec_ids:
134                        execution_counter = new_execution_counter
135                        self.current_execution_id = new_execution_id
136
137                        start_time = datetime.fromisoformat(parsed["timestamp"])
138                        self.db.add_execution(
139                            execution_id=self.current_execution_id,
140                            start_time=start_time,
141                            log_file=str(log_path),
142                            description=f"Execution {execution_counter}",
143                        )
144                        existing_exec_ids.add(self.current_execution_id)
145                    else:
146                        execution_counter = new_execution_counter
147                        self.current_execution_id = new_execution_id
148
149                if not self.current_execution_id:
150                    self.current_execution_id = f"{self.current_date}_exec000"
151                    if self.current_execution_id not in existing_exec_ids:
152                        self.db.add_execution(
153                            execution_id=self.current_execution_id,
154                            start_time=datetime.fromisoformat(
155                                f"{self.current_date} 00:00:00",
156                            ),
157                            log_file=str(log_path),
158                            description="Execution 0",
159                        )
160                        existing_exec_ids.add(self.current_execution_id)
161
162                if not self.db.log_entry_exists(
163                    parsed["timestamp"],
164                    parsed["file_name"],
165                    parsed["line_number"],
166                ):
167                    self.db.add_log_entry(
168                        execution_id=self.current_execution_id,
169                        **parsed,
170                    )
171                    entries_indexed += 1
172
173        return entries_indexed

 62        self.current_date = self.log_file_path.stem
 63
 64    def _initialize_execution(self) -> None:
 65        """Initialize a new execution for this logging session."""
 66        if not self.db:
 67            return
 68
 69        existing_executions = self.db.get_executions(str(self.log_file_path))
 70
 71        exec_num = 0
 72        if existing_executions:
 73            last_exec_id = existing_executions[-1]["execution_id"]
 74            exec_num = int(last_exec_id.split("_exec")[-1])
 75
 76        self.current_execution_id = f"{self.current_date}_exec{exec_num:03d}"
 77
 78        start_time = datetime.datetime.now()
 79        self.db.add_execution(
 80            execution_id=self.current_execution_id,
 81            start_time=start_time,
 82            log_file=str(self.log_file_path),
 83            description=f"Execution {exec_num}",
 84        )
 85
 86    @staticmethod
 87    def _should_exclude_record(record: LogRecord) -> bool:
 88        """Check if a log record should be excluded from database.
 89
 90        Args:
 91            record (LogRecord): The log record to check
 92
 93        Returns:
 94            bool: True if the record should be excluded
 95        """
 96        pathname = record.pathname.replace("\\", "/")
 97        return any(pattern in pathname for pattern in [".venv", "site-packages"])

100    def emit(self, record: LogRecord) -> None:
101        """Emit a log record to the database.
102
103        Args:
104            record (LogRecord): The log record to emit
105        """
106        if not self.db or not self.current_execution_id:
107            return
108
109        if self._should_exclude_record(record):
110            return
111
112        try:
113            message = record.getMessage()
114
115            if detect_execution_boundary(message):
116                existing_executions = self.db.get_executions(str(self.log_file_path))
117                exec_num = len(existing_executions)
118                self.current_execution_id = f"{self.current_date}_exec{exec_num:03d}"
119
120                start_time = datetime.datetime.fromtimestamp(record.created)
121                self.db.add_execution(
122                    execution_id=self.current_execution_id,
123                    start_time=start_time,
124                    log_file=str(self.log_file_path),
125                    description=f"Execution {exec_num}",
126                )
127
128            raw_line = self.format(record)
129            parsed = parse_log_line(raw_line, current_date=self.current_date)
130            if not parsed:
131                return
132            self.db.add_log_entry(
133                execution_id=self.current_execution_id,
134                **parsed,
135            )
136
137        except Exception:  # noqa: BLE001
138            self.handleError(record)
139
140    @override
141    def close(self) -> None:
142        """Close the database handler and update execution end time."""
143        if self.db and self.current_execution_id:
144            self.db.update_execution_end_time(
145                self.current_execution_id,
146                datetime.datetime.now(),
147            )
148
149            self.db.close()
150            self.db = None
151
152        super().close()

Logging handler that writes to SQLite database in real-time.

This handler can be attached to any Python logger (including loggerplusplus loggers) to automatically write all log messages to a SQLite database as they are generated.

The database structure is compatible with the log_manager indexer, so you can use all the existing query and analysis tools.

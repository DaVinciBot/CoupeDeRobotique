176    def get_go_to_position(
177        self,
178        ally_position: OrientedPoint,
179        team_color: TeamColor,
180    ) -> OrientedPoint | Point | None:
181        """Determine the best go-to position for an ally in the given zone.
182
183        Args:
184            ally_position (OrientedPoint): Position of the ally.
185            team_color (TeamColor): Color of the team.
186
187        Returns:
188            OrientedPoint | Point | None:
189                Best go-to position, or ``None`` if inaccessible.
190        """
191        if not self.is_accessible(team_color):
192            self.logger.debug(
193                f"GoTo position request: Zone {self.zone_type} is not accessible.",
194            )
195            return None
196
197        # If no specific go-to positions are defined, return the centroid of the zone
198        if not self.go_to_positions:
199            self.logger.debug(
200                "GoTo position request: "
201                f"No defined go-to positions for zone {self.zone_type}, "
202                f"returning centroid [{self.polygon.centroid}]",
203            )
204            return self.polygon.centroid
205
206        nearest_position = min(
207            self.go_to_positions,
208            key=ally_position.distance,
209        )
210        msg = (
211            "GoTo position request: Nearest go-to position to ally "
212            f"[{ally_position}] is [{nearest_position}]"
213        )
214        self.logger.debug(msg)
215
216        return nearest_position

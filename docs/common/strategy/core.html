 18    def __init__(
 19        self,
 20        start: BaseTaskNode,
 21        logger: Logger | None = None,
 22        *,
 23        parallel: bool = False,
 24    ) -> None:
 25        """Create a new :class:``GraphRunner``.
 26
 27        Args:
 28            start (BaseTaskNode): The entry node for the graph.
 29            logger (Logger | None, optional):
 30                Logger instance for debugging. Defaults to None.
 31            parallel (bool, optional): Execute all valid transitions in parallel when
 32                ``True``. Defaults to ``False``.
 33        """
 34        self._logger = logger or Logger(
 35            identifier="GraphRunner",
 36            follow_logger_manager_rules=True,
 37        )
 38        self.parallel = parallel
 39        self.active: list[BaseTaskNode] = [start]
 40        self.prev: dict[BaseTaskNode, BaseTaskNode | None] = {start: None}
 41        self._logger.info(
 42            f"[STRAT] GraphRunner initialized with start: '{start.name}', "
 43            f"parallel={self.parallel}",
 44        )
 45
 46    def handle(self, ctx: BaseGameContext) -> None:
 47        """Advance the graph execution by one step.
 48
 49        Args:
 50            ctx (BaseGameContext): Context passed to each node.
 51        """
 52        if not self.active:
 53            self._logger.warning(
 54                "[STRAT] No active nodes - execution complete or not started",
 55            )
 56            return
 57
 58        next_active: list[BaseTaskNode] = []
 59        for node in self.active:
 60            prev_node = self.prev.get(node)
 61
 62            # Log entry if first time
 63            if not node.entered:
 64                task_name = node.tasks[0].__class__.__name__ if node.tasks else "NoTask"
 65                self._logger.info(f"[STRAT] ==> Entering: {node.name} [{task_name}]")
 66
 67            done = node.handle(ctx)
 68            if not done:
 69                self._logger.debug(f"[STRAT] ... executing: {node.name}")
 70                next_active.append(node)
 71                continue
 72
 73            # Node completed
 74            self._logger.info(
 75                f"[STRAT] <== Finished: {node.name} with status {node.status.name}",
 76            )
 77
 78            # Gather valid transitions
 79            valid_transitions = [
 80                t
 81                for t in node.transitions
 82                if t.can_transit(from_node=prev_node, ctx=ctx)
 83                and t.target not in self.prev.items()
 84            ]
 85            if not valid_transitions:
 86                msg = (
 87                    f"[STRAT]     No valid transitions from '{node.name}' - branch ends"
 88                )
 89                self._logger.info(msg)
 90                continue
 91
 92            if self.parallel:
 93                for transition in valid_transitions:
 94                    target = transition.target
 95                    msg = (
 96                        f"[STRAT]     Transition: '{node.name}' -> '{target.name}' "
 97                        f"via {transition.__class__.__name__}"
 98                    )
 99                    self._logger.info(msg)
100                    next_active.append(target)
101                    self.prev[target] = node
102            else:
103                # Choose the transition leading to the highest-scoring node
104                best = max(
105                    valid_transitions,
106                    key=lambda t, prev=prev_node: t.target.score(prev, ctx),
107                )
108                score_val = best.target.score(prev_node, ctx)
109                target = best.target
110                msg = (
111                    f"[STRAT]     Chosen: '{node.name}' -> '{target.name}' "
112                    f"via {best.__class__.__name__} (score={score_val:.2f})"
113                )
114                self._logger.info(msg)
115                next_active.append(target)
116                self.prev[target] = node
117
118        self.active = next_active
119
120    def run(self, ctx: BaseGameContext, max_steps: int = 1000) -> None:
121        """Run the graph until completion or until ``max_steps`` iterations.
122
123        Args:
124            ctx (BaseGameContext): Game context passed to task nodes.
125            max_steps (int, optional):
126                Safety limit to prevent infinite loops. Defaults to 1000.
127        """
128        step = 0
129        while self.active and step < max_steps:
130            active_names = [n.name for n in self.active]
131            msg = f"[STRAT] Step {step + 1}, active: {active_names}"
132            self._logger.debug(msg)
133            self.handle(ctx)
134            step += 1
135        if self.active:
136            remaining = [n.name for n in self.active]
137            msg = (
138                f"[STRAT] Max steps reached ({max_steps}) - "
139                f"active nodes remaining: {remaining}"
140            )
141            self._logger.warning(msg)
142        else:
143            self._logger.info("[STRAT] All nodes completed")

 17class SubGraphBuilder:
 18    """Build a :class:`BaseSubGraph` from nodes and transitions.
 19
 20    The builder can merge subgraphs and produce a standalone subgraph ready for
 21    execution.
 22    """
 23
 24    def __init__(self) -> None:
 25        """Initialize empty builder and logger."""
 26        self._logger = Logger(
 27            identifier="SubGraphBuilder",
 28            follow_logger_manager_rules=True,
 29        )
 30        self.nodes: dict[str, BaseTaskNode] = {}
 31        self._transitions: list[tuple[str, BaseTransition]] = []
 32        self._logger.info("[STRAT:Builder] Initialized")
 33
 34    def add_node(self, name: str, node: BaseTaskNode) -> SubGraphBuilder:
 35        """Register a task node.
 36
 37        Args:
 38            name (str): Name for the node; must be unique.
 39            node (BaseTaskNode): Node instance to register.
 40
 41        Returns:
 42            SubGraphBuilder: ``self`` to allow call chaining.
 43
 44        Raises:
 45            KeyError: If ``name`` already exists in the builder.
 46        """
 47        if name in self.nodes:
 48            msg = f"[STRAT:Builder] Node '{name}' already registered"
 49            self._logger.error(msg)
 50            raise KeyError(msg)
 51        self.nodes[name] = node
 52        self._logger.debug(f"[STRAT:Builder] Added node: '{name}'")
 53        return self
 54
 55    def connect(self, from_name: str, transition: BaseTransition) -> SubGraphBuilder:
 56        """Queue a transition from an existing node.
 57
 58        Args:
 59            from_name (str): Name of the source node.
 60            transition (BaseTransition): Transition to append.
 61
 62        Returns:
 63            SubGraphBuilder: ``self`` for chaining.
 64
 65        Raises:
 66            KeyError: If ``from_name`` is not registered.
 67        """
 68        if from_name not in self.nodes:
 69            msg = f"[STRAT:Builder] Source node '{from_name}' not found"
 70            self._logger.error(msg)
 71            raise KeyError(msg)
 72        self._transitions.append((from_name, transition))
 73        self._logger.debug(
 74            f"[STRAT:Builder] Queued transition: '{from_name}' -> {transition}",
 75        )
 76        return self
 77
 78    def add_subgraph(self, subgraph: BaseSubGraph, prefix: str = "") -> SubGraphBuilder:
 79        """Merge another subgraph into this builder.
 80
 81        Args:
 82            subgraph (BaseSubGraph): The subgraph to merge.
 83            prefix (str, optional): Prefix for new node names. Defaults to "".
 84
 85        Returns:
 86            SubGraphBuilder: ``self`` for chaining.
 87        """
 88        mapping: dict[BaseTaskNode, BaseTaskNode] = {}
 89        for old in subgraph.get_all_nodes():
 90            new_name = prefix + old.name
 91            new_node = type(old)(
 92                name=new_name,
 93                tasks=old.tasks,
 94                scoring_function=old.scoring_function,
 95            )
 96            self.add_node(new_name, new_node)
 97            mapping[old] = new_node
 98        for old in subgraph.get_all_nodes():
 99            from_new = mapping[old]
100            for t in old.transitions:
101                if t.target not in mapping:
102                    continue
103                new_transition = type(t)(mapping[t.target])
104                from_new.add_transition(new_transition)
105                self._logger.debug(
106                    f"[STRAT:Builder] Recreated transition: '{from_new.name}' -> "
107                    f"'{mapping[t.target].name}'",
108                )
109
110        return self
111
112    def build(
113        self,
114        entry: str | BaseTaskNode,
115        exits: str | BaseTaskNode | list[str | BaseTaskNode],
116    ) -> BaseSubGraph:
117        """Finalize construction and return a :class:``BaseSubGraph``.
118
119        Args:
120            entry (str | BaseTaskNode): Entry node or its name.
121            exits (str | BaseTaskNode | list[str | BaseTaskNode]): One or more
122                exit nodes or their names.
123
124        Returns:
125            BaseSubGraph: The assembled subgraph ready for execution.
126
127        Raises:
128            KeyError: If ``entry`` or any ``exits`` are not registered.
129        """
130        # Resolve entry
131        entry_node = self._resolve(entry)
132        # Resolve exits
133        if isinstance(exits, list):
134            exit_nodes = [self._resolve(e) for e in exits]
135        else:
136            exit_nodes = [self._resolve(exits)]
137        # Apply transitions
138        for from_name, transition in self._transitions:
139            node = self.nodes[from_name]
140            node.add_transition(transition)
141            self._logger.debug(
142                "[STRAT:Builder] Connected: "
143                f"'{from_name}' -> '{transition.target.name}'",
144            )
145        # Validate
146        missing = [n for n in exit_nodes if n.name not in self.nodes]
147        if missing:
148            msg = (
149                "[STRAT:Builder] Exit nodes not registered: "
150                f"{[n.name for n in missing]}"
151            )
152            self._logger.error(msg)
153            raise KeyError(msg)
154        exits_names = [n.name for n in exit_nodes]
155        self._logger.info(
156            "[STRAT:Builder] Building subgraph: "
157            f"entry='{entry_node.name}', exits={exits_names}",
158        )
159        return BaseSubGraph(
160            entry_node=entry_node,
161            exit_nodes=exit_nodes,
162            all_nodes=list(self.nodes.values()),
163        )
164
165    def _resolve(self, item: str | BaseTaskNode) -> BaseTaskNode:
166        if isinstance(item, str):
167            if item not in self.nodes:
168                msg = f"[STRAT:Builder] Node '{item}' not found"
169                self._logger.error(msg)
170                raise KeyError(msg)
171            return self.nodes[item]
172        return item

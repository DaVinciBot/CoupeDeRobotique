158        """Execute a single task and record its result.
159
160        Args:
161            idx (int): Index of the task to execute.
162            ctx (BaseGameContext): The current game context.
163        """

180            self.task_done[idx] = True

183            )
184        except Exception as e:  # noqa: BLE001
185            self.exceptions[idx] = e
186            self.task_done[idx] = True
187            self._logger.error(

193        """Execute tasks sequentially.
194
195        Args:
196            ctx (BaseGameContext): The current game context.
197
198        Returns:
199            bool: ``True`` when all tasks are completed, ``False`` otherwise.
200        """
201        if self.status in {TaskStatus.DONE, TaskStatus.FAILED, TaskStatus.TIMEOUT}:
202            self._logger.debug(
203                f"[STRAT:Task] '{self.name}' already completed: {self.status.name}",
204            )
205            return True
206
207        if self.start_time is None:
208            self.start_time = time.time()
209            self.status = TaskStatus.IN_PROGRESS
210            self._logger.info(f"[STRAT:Task] Started execution of '{self.name}'")
211
212        # Handle only the first incomplete task per cycle
213        try:
214            next_idx = next(i for i, done in enumerate(self.task_done) if not done)
215        except StopIteration:
216            next_idx = None
217
218        if next_idx is not None:
219            self._handle_task(next_idx, ctx)
220
221        if all(self.task_done):
222            self.end_time = time.time()
223            any_failed = any(
224                (ex is not None) and not isinstance(ex, TimeoutError)
225                for ex in self.exceptions
226            )
227            any_timeout = any(
228                isinstance(ex, TimeoutError) for ex in self.exceptions if ex is not None
229            )
230            if any_failed:
231                self.status = TaskStatus.FAILED
232            elif any_timeout:
233                self.status = TaskStatus.TIMEOUT
234            else:
235                self.status = TaskStatus.DONE
236
237            elapsed = self.end_time - self.start_time
238            self._logger.info(
239                f"[STRAT:Task] Finished '{self.name}' {self.status.name} "
240                f"in {elapsed:.2f}s",
241            )
242            return True
243
244        return False

 47    def on_timeout(self, _ctx: BaseGameContext) -> None:
 48        """Hook called once when the timeout is reached.
 49
 50        Args:
 51            _ctx (BaseGameContext): Current game context.
 52
 53        This method can be overridden in subclasses to implement custom
 54        behaviour when the timeout triggers.
 55        """
 56        self._logger.warning(
 57            f"[STRAT] Timeout reached: '{self.name}' ({self.timeout_seconds:.2f}s)",
 58        )
 59
 60    @override
 61    def execute(self, ctx: BaseGameContext) -> bool:
 62        """Execute tasks but enforce a maximum duration.
 63
 64        Args:
 65            ctx (BaseGameContext): Game context used for task execution.
 66
 67        Returns:
 68            bool: ``True`` if all tasks completed or a timeout occurred,
 69            ``False`` otherwise.
 70        """
 71        # If already completed, no-op
 72        if self.status in {TaskStatus.DONE, TaskStatus.FAILED, TaskStatus.TIMEOUT}:
 73            self._logger.debug(
 74                f"[STRAT] TimeoutTaskNode '{self.name}' "
 75                f"already completed: {self.status.name}",
 76            )
 77            return True
 78
 79        now = time.time()
 80
 81        # On first run, start timer and status
 82        if self.start_time is None:
 83            self.start_time = now
 84            self.status = TaskStatus.IN_PROGRESS
 85            self._logger.info(
 86                f"[STRAT] Started TimeoutTaskNode: '{self.name}' "
 87                f"(timeout: {self.timeout_seconds:.2f}s)",
 88            )
 89
 90        elapsed = now - self.start_time
 91        self._logger.debug(
 92            f"[STRAT] Node '{self.name}' "
 93            f"elapsed: {elapsed:.2f}s / {self.timeout_seconds:.2f}s",
 94        )
 95
 96        # Check for timeout
 97        if elapsed >= self.timeout_seconds:
 98            if not self._timeout_triggered:
 99                self.on_timeout(ctx)
100                self._timeout_triggered = True
101            self.status = TaskStatus.TIMEOUT
102            self.end_time = now
103            self._logger.info(
104                f"[STRAT] TimeoutTaskNode '{self.name}' status: TIMEOUT",
105            )
106            return True
107
108        # Delegate to base execution if within timeout
109        return super().execute(ctx)

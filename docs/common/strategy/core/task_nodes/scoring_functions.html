28    @override
29    def compute(
30        self,
31        prev_node: BaseTaskNode,
32        current_node: BaseTaskNode,
33        ctx: BaseGameContext,
34    ) -> float:
35        """Compute the efficiency score between point gain and time taken.
36
37        Args:
38            prev_node (BaseTaskNode): _description_
39            current_node (BaseTaskNode): _description_
40            ctx (BaseGameContext): _description_
41
42        Returns:
43            float: _description_
44        """
45        estimated_duration = (
46            current_node.estimated_duration(ctx)
47            if callable(current_node.estimated_duration)
48            else current_node.estimated_duration
49        )
50        points = (
51            current_node.points(ctx)
52            if callable(current_node.points)

class NavigationScoringFunction(common.strategy.core.task_nodes.scoring_functions.BaseScoringFunction): 
17class NavigationScoringFunction(BaseScoringFunction):
18    """Score nodes higher the closer the robot is to a goal."""
19
20    def __init__(self, goal: int) -> None:
21        """Initialize the navigation scoring function.
22
23        Args:
24            goal (int): The navigation goal to reach.
25        """
26        self.goal = goal
27
28    @override
29    def compute(
30        self,
31        prev_node: BaseTaskNode,
32        current_node: BaseTaskNode,
33        ctx: BaseGameContext,
34    ) -> float:
35        """Compute the navigation score based on distance to the goal.
36
37        Args:
38            prev_node (BaseTaskNode): The previous task node.
39            current_node (BaseTaskNode): The current task node.
40            ctx (BaseGameContext): The game context.
41
42        Returns:
43            float: The computed navigation score.
44        """
45        distance = ctx.arena.ally_zone.point.distance(
46            ctx.arena.compute_goal_position(self.goal),
47        )
48        return max(0.0, (1.0 - distance / 200.0) * 6.0)

 75        """Compute a score for this node.

116    def _handle_task(self, idx: int, ctx: BaseGameContext) -> None:
117        """Execute a single task and record its result.
118
119        Args:
120            idx (int): Index of the task to execute.
121            ctx (BaseGameContext): The current game context.
122        """
123        task = self.tasks[idx]
124        self.logger.debug(f"Handling task {idx} of node '{self.name}'")
125        try:
126            done = task.handle(ctx)
127            self.results[idx] = done
128            if done:
129                self.task_done[idx] = True
130                self.logger.info(
131                    f"Task {idx} in node '{self.name}' completed successfully",
132                )
133            else:
134                self.logger.debug(
135                    f"Task {idx} in node '{self.name}' not done yet",
136                )
137        except TimeoutError as e:
138            self.exceptions[idx] = e
139            self.task_done[idx] = True
140            self.logger.warning(
141                f"Task {idx} in node '{self.name}' timed out: {e}",
142            )
143        except Exception as e:  # noqa: BLE001
144            self.exceptions[idx] = e
145            self.task_done[idx] = True
146            self.logger.error(
147                f"Task {idx} in node '{self.name}' failed: {e}', "
148                f"traceback {traceback.format_exc()}",
149            )
150
151    def execute(self, ctx: BaseGameContext) -> bool:
152        """Execute tasks sequentially.
153
154        Args:
155            ctx (BaseGameContext): The current game context.
156
157        Returns:
158            bool: ``True`` when all tasks are completed, ``False`` otherwise.
159        """
160        if self.status in {TaskStatus.DONE, TaskStatus.FAILED, TaskStatus.TIMEOUT}:
161            self.logger.debug(
162                f"Node '{self.name}' already completed with status {self.status.name}",
163            )
164            return True
165
166        if self.start_time is None:
167            self.start_time = time.time()
168            self.status = TaskStatus.IN_PROGRESS
169            self.logger.info(f"Started execution of node '{self.name}'")
170
171        # Handle only the first incomplete task per cycle
172        try:
173            next_idx = next(i for i, done in enumerate(self.task_done) if not done)
174        except StopIteration:
175            next_idx = None
176
177        if next_idx is not None:
178            self._handle_task(next_idx, ctx)
179
180        if all(self.task_done):
181            self.end_time = time.time()
182            any_failed = any(
183                (ex is not None) and not isinstance(ex, TimeoutError)
184                for ex in self.exceptions
185            )
186            any_timeout = any(
187                isinstance(ex, TimeoutError) for ex in self.exceptions if ex is not None
188            )
189            if any_failed:
190                self.status = TaskStatus.FAILED
191            elif any_timeout:
192                self.status = TaskStatus.TIMEOUT
193            else:
194                self.status = TaskStatus.DONE
195
196            elapsed = self.end_time - self.start_time
197            self.logger.info(
198                f"Finished node '{self.name}' with status {self.status.name}"
199                f" in {elapsed:.2f}s",
200            )
201            return True
202
203        return False
204
205    def handle(self, ctx: BaseGameContext) -> bool:
206        """Enter the node, execute tasks, and exit when finished.
207
208        Args:
209            ctx (BaseGameContext): The current game context.
210
211        Returns:
212            bool: ``True`` when all tasks are completed, ``False`` otherwise.
213        """
214        if not self.entered:
215            self.on_enter(None, ctx)
216            self.entered = True
217
218        done = self.execute(ctx)
219
220        if done and not self._exited:
221            self.on_exit(None, ctx)
222            self._exited = True
223
224        return done

 82                for t in node.transitions
 83                if t.can_transit(from_node=prev_node, ctx=ctx)
 84                and t.target not in self.prev.items()

106                )

 55            self.logger.warning(msg)

 77            self.logger.info(msg)
 78

 81                t
 82                for t in node.transitions
 83                if t.can_transit(from_node=prev_node, ctx=ctx)
 84                and t.target not in self.prev.items()
 85            ]

 87                msg = f"    No valid transitions from '{node.name}'; branch ends here."
 88                self.logger.info(msg)

 95                        f"    Transition: '{node.name}' -> '{target.name}' via "

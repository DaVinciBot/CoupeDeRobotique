 15    """Speed profile with acceleration, cruise, and deceleration phases."""
 16
 17    def __init__(
 18        self,
 19        acceleration: float,
 20        max_speed: float,
 21        deceleration: float,
 22    ) -> None:
 23        """Initialize with acceleration, max speed, and deceleration.
 24
 25        Args:
 26            acceleration (float):
 27                The constant acceleration during the acceleration phase.
 28            max_speed (float): The maximum allowed cruising speed.
 29            deceleration (float):
 30                The constant deceleration during the deceleration phase.
 31        """
 32        super().__init__(max_speed=max_speed)
 33        self.acceleration = acceleration
 34        self.deceleration = deceleration
 35
 36    def _compute_trapezoidal_params(
 37        self,
 38        departure_speed: float,
 39        arrival_speed: float,
 40    ) -> tuple[float, float, float, float]:
 41        """Computes trapezoidal motion parameters.
 42
 43        Args:
 44            departure_speed (float): Speed at the beginning of the motion.
 45            arrival_speed (float): Speed at the end of the motion.
 46
 47        Returns:
 48            tuple[float, float, float, float]: Time to accelerate, time to
 49                decelerate, distance during acceleration and distance during
 50                deceleration.
 51        """
 52        t_acc = (self._max_speed - departure_speed) / self.acceleration
 53        t_dec = (self._max_speed - arrival_speed) / self.deceleration
 54        d_acc = (self._max_speed**2 - departure_speed**2) / (2 * self.acceleration)
 55        d_dec = (self._max_speed**2 - arrival_speed**2) / (2 * self.deceleration)
 56        return t_acc, t_dec, d_acc, d_dec
 57
 58    def _compute_triangular_peak_speed(
 59        self,
 60        distance: float,
 61        departure_speed: float,
 62        arrival_speed: float,
 63    ) -> float:
 64        """Calculate peak speed for triangular profile when no cruise phase.
 65
 66        Args:
 67            distance (float): Total distance to travel.
 68            departure_speed (float): Speed at the start.
 69            arrival_speed (float): Speed at the end.
 70
 71        Returns:
 72            float: Peak speed reached in the triangular profile.
 73        """
 74        v_peak_denom = (1 / (2 * self.acceleration)) + (1 / (2 * self.deceleration))
 75        v_peak_sq = (
 76            (departure_speed**2 / (2 * self.acceleration))
 77            + (arrival_speed**2 / (2 * self.deceleration))
 78            + distance
 79        ) / v_peak_denom
 80        return sqrt(v_peak_sq)
 81
 82    def _speed_time(
 83        self,
 84        time_elapsed: float,
 85        t_acc: float,
 86        t_dec: float,
 87        departure_speed: float,
 88        arrival_speed: float,
 89    ) -> float:
 90        if time_elapsed < t_acc:
 91            return departure_speed + self.acceleration * time_elapsed
 92        if time_elapsed < t_acc + t_dec:
 93            return max(
 94                arrival_speed + self.deceleration * (t_acc + t_dec - time_elapsed),
 95                arrival_speed,
 96            )
 97        return arrival_speed
 98
 99    def _speed_trapezoidal(
100        self,
101        time_elapsed: float,
102        distance: float,
103        t_acc: float,
104        t_dec: float,
105        d_acc: float,
106        d_dec: float,
107        departure_speed: float,
108        arrival_speed: float,
109    ) -> float:
110        d_cruise = distance - (d_acc + d_dec)
111        t_cruise = d_cruise / self._max_speed
112        t_total = t_acc + t_cruise + t_dec
113        if time_elapsed < t_acc:
114            return departure_speed + self.acceleration * time_elapsed
115        if time_elapsed < t_acc + t_cruise:
116            return self._max_speed
117        if time_elapsed < t_total:
118            return max(
119                self._max_speed - self.deceleration * (time_elapsed - t_acc - t_cruise),
120                arrival_speed,
121            )
122        return arrival_speed

124    def _speed_triangular(
125        self,
126        time_elapsed: float,
127        distance: float,
128        departure_speed: float,
129        arrival_speed: float,
130    ) -> float:
131        v_peak = self._compute_triangular_peak_speed(
132            distance,
133            departure_speed,
134            arrival_speed,
135        )
136        t_peak = (v_peak - departure_speed) / self.acceleration
137        t_decel = (v_peak - arrival_speed) / self.deceleration
138        t_total = t_peak + t_decel
139        if time_elapsed < t_peak:
140            return departure_speed + self.acceleration * time_elapsed
141        if time_elapsed < t_total:
142            return max(
143                v_peak - self.deceleration * (time_elapsed - t_peak),
144                arrival_speed,
145            )
146        return arrival_speed

148    def _distance_time(
149        self,
150        time_elapsed: float,
151        t_acc: float,
152        t_dec: float,
153        d_acc: float,
154        d_dec: float,
155        departure_speed: float,
156        arrival_speed: float,
157    ) -> float:
158        t_total = t_acc + t_dec
159        if time_elapsed < t_acc:
160            return (
161                departure_speed * time_elapsed

 67    @BaseAvoidance.ensure_original_task_storage
 68    def handle(
 69        self,
 70        current_navigator_task: NavigatorTask,
 71        ally_zone: AllyZone,
 72        enemy_zone: EnemyZone,
 73    ) -> TrajectoryPlanCommand:
 74        """Process avoidance logic based on current zones and navigation state.
 75
 76        Args:
 77            current_navigator_task (NavigatorTask):
 78                The current navigation task instance.
 79            ally_zone (AllyZone): Ally zone providing positional data.
 80            enemy_zone (EnemyZone): Enemy zone used for obstacle detection.
 81
 82        Returns:
 83            TrajectoryPlanCommand:
 84            The trajectory command after processing avoidance logic.
 85        """
 86        position: OrientedPoint = ally_zone.point
 87        self.logger.debug(
 88            f"Handling avoidance at position: {position}, current state: {self.state}",
 89        )
 90
 91        # 1. Timeout check
 92        if self._has_timed_out():
 93            self.logger.warning("Avoidance timed out. Aborting task.")
 94            return self._abort(current_navigator_task, position)
 95
 96        # 2. Obstacle detected → init backward task
 97        if (
 98            self.acs_detector.is_acs_triggered(ally_zone, enemy_zone)
 99            and self.state == AvoidanceState.IDLE
100        ):
101            self.logger.info(
102                "Obstacle detected. starting backward avoidance. "
103                f"Distance: {ally_zone.point.distance(enemy_zone.point)}",
104            )
105
106            # Create backward navigator task
107            self.backward_navigator_task = NavigatorTask(
108                NavigatorTaskParams(
109                    goal=None,
110                    timeout=None,
111                    stabilization_delay=0.0,
112                    path_planner_params=DeltaPathPlannerParams(
113                        distance=self.params.backward_distance,
114                    ),
115                    trajectory_planner_params=SequentialTrajectoryPlannerParams(
116                        direction=Direction.BACKWARD,
117                    ),
118                    speed_profiler=self.params.backward_speed_profiler,
119                    avoidance_params=NoAvoidanceParams(),
120                    acs_detection_profile_params=NoAcsDetectionProfileParams(),
121                ),
122            )
123
124            self.state = AvoidanceState.AVOIDING
125            current_navigator_task.state = NavigatorTaskState.AVOIDING
126            self._start_timer()
127            self.logger.debug(f"Timer started at: {self._avoiding_start_time}")
128
129            # Execute first backward command immediately
130            cmd = self.backward_navigator_task.handle(ally_zone, enemy_zone)
131            current_navigator_task.current_trajectory_command = cmd
132            return cmd
133
134        # 3. While avoiding → keep executing backward
135        if (
136            self.state == AvoidanceState.AVOIDING
137            and self.backward_navigator_task is not None
138        ):
139            self.logger.info("Executing backward avoidance maneuver.")
140            cmd: TrajectoryPlanCommand = self.backward_navigator_task.handle(
141                ally_zone,
142                enemy_zone,
143            )
144
145            # Clear the backward task if it has finished
146            if self.backward_navigator_task.state == NavigatorTaskState.FINISHED:
147                self.backward_navigator_task = None
148
149            current_navigator_task.current_trajectory_command = cmd
150            return cmd  # Return the command from the backward avoidance task
151
152        # 4. Finished backward, obstacle clear → replan
153        if (
154            self.state == AvoidanceState.AVOIDING
155            and self.backward_navigator_task is None
156        ):
157            self.logger.info("Obstacle cleared. Replanning trajectory.")
158
159            # Obstacle is no longer detected, replan from current position
160            last_params = cast(
161                "BasePathPlannerPlanPathParams",
162                current_navigator_task.path_planner.last_plan_path_params,
163            )
164            last_params.start = position  # Update start position to current location
165
166            self.logger.debug(f"Replanning from updated start: {position}")
167
168            new_path = current_navigator_task.path_planner.plan_path(last_params)
169            current_navigator_task.trajectory_planner.plan_trajectory(new_path)
170            current_navigator_task.trajectory_planner.start_planning()
171
172            self.logger.debug("Trajectory planner reset internal clock.")
173            # reset timer just for logging/manure measurement
174            self._reset_timer()
175            self.logger.debug("Timer reset after avoidance completion.")
176
177            self.state = AvoidanceState.IDLE
178            current_navigator_task.state = NavigatorTaskState.IN_PROGRESS
179
180            self.logger.info("Avoidance complete. Resuming normal operation.")
181            return cast(
182                "TrajectoryPlanCommand",
183                current_navigator_task.current_trajectory_command,
184            )  # Avoidance complete, continue as normal
185
186        # 4. Continue with current command
187        self.logger.debug(
188            "No avoidance action required. Continuing original trajectory.",
189        )
190        return cast(
191            "TrajectoryPlanCommand",
192            current_navigator_task.current_trajectory_command,
193        )

15class BackAvoidanceParams(BaseAvoidanceParams):
16    """Configuration for a backward avoidance strategy.
17
18    This strategy stops the system when an obstacle is detected within a
19    specified distance and waits for a defined timeout period before taking
20    further action.
21    """
22
23    def __init__(
24        self,
25        timeout: float,
26        backward_distance: float,
27        backward_speed_profiler: SpeedProfiler,
28    ) -> None:
29        """Initialize parameters for backward avoidance.
30
31        Args:
32            timeout (float):
33                Time to wait after stopping before checking again in seconds.
34            backward_distance (float):
35                Distance to reverse when avoidance is triggered.
36            backward_speed_profiler (SpeedProfiler):
37                Profiler for the backward motion.
38        """
39        self.backward_distance: float = backward_distance
40        self.backward_speed_profiler: SpeedProfiler = backward_speed_profiler
41        super().__init__(AvoidanceStrategy.BACK, timeout * 1000.0)

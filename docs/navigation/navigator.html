 49        Returns:

 51        """
 52        if self._tasks_queue:

 54                params=self._tasks_queue.popleft(),

 56            self.logger.info(f"Switched to new task: {self.current_task}")
 57            return True
 58        self.current_task = None
 59        return False

 67        """Add a navigation task to the queue.
 68

 70            navigator_task_params (NavigatorTaskParams):
 71                The parameters for the navigation task.
 72            skip_queue (bool, optional):
 73                If ``True``, skip the queue and execute the task immediately.
 74                Defaults to ``False``.
 75        """
 76        if skip_queue:

 32    def __init__(self, params: NavigatorTaskParams) -> None:
 33        """Initialize the NavigatorTask.
 34
 35        Args:
 36            params (NavigatorTaskParams): The parameters for the navigation task.
 37        """
 38        from navigation.avoidance.avoidance_factory import (  # noqa: PLC0415
 39            AvoidanceFactory,
 40        )
 41
 42        self.params = params
 43        self.path_planner: BasePathPlanner = PathPlannerFactory.instantiate(
 44            params.path_planner_params,
 45        )
 46        self.trajectory_planner: BaseTrajectoryPlanner = (
 47            TrajectoryPlannerFactory.instantiate(
 48                params.trajectory_planner_params,
 49                params.speed_profiler,
 50            )
 51        )
 52        self.avoidance: BaseAvoidance = AvoidanceFactory.instantiate(
 53            params.avoidance_params,
 54            params.acs_detection_profile_params,
 55        )
 56        self.current_trajectory_command: TrajectoryPlanCommand | None = None
 57        self.state: NavigatorTaskState = NavigatorTaskState.NOT_PLANNED
 58        self._start_time: float | None = None
 59        self._stabilization_start_time: float | None = None
 60
 61    def _get_elapsed_time(self) -> float:
 62        if self._start_time is None:
 63            return 0.0
 64        return time.time() - self._start_time
 65
 66    def _get_stabilization_elapsed(self) -> float:
 67        if self._stabilization_start_time is None:
 68            return 0.0
 69        return time.time() - self._stabilization_start_time
 70
 71    def _plan_task(self, ally_zone: AllyZone) -> None:
 72        self._start_time = time.time()
 73        plan_path_params: BasePathPlannerPlanPathParams = (
 74            PathPlannerPathPlanParamsFactory.instantiate(
 75                strategy=self.params.path_planner_params.path_finding_strategy,
 76                current_position=ally_zone.point,
 77                goal=cast("OrientedPoint", self.params.goal),
 78            )
 79        )
 80        path: list[OrientedPoint] = self.path_planner.plan_path(plan_path_params)
 81        self.trajectory_planner.plan_trajectory(path)
 82        self.current_trajectory_command = self.trajectory_planner.get_plan()
 83        self.state = NavigatorTaskState.IN_PROGRESS
 84
 85    def _has_timed_out(self) -> bool:
 86        return (
 87            False
 88            if self.params.timeout is None or self._start_time is None
 89            else self._get_elapsed_time() > self.params.timeout
 90        )
 91
 92    def _abort(self) -> TrajectoryPlanCommand:
 93        self.state = NavigatorTaskState.ABORT
 94        self.current_trajectory_command = TrajectoryPlanCommand.create_stop_command(
 95            current_position=cast("OrientedPoint", self.params.goal),
 96        )
 97        return self.current_trajectory_command
 98
 99    def _is_finished(self) -> bool:
100        if self._start_time is None or self.state == NavigatorTaskState.FINISHED:
101            return False
102        return self._get_elapsed_time() > self.trajectory_planner.get_total_duration()
103
104    def handle(
105        self,
106        ally_zone: AllyZone,
107        enemy_zone: EnemyZone,
108    ) -> TrajectoryPlanCommand:
109        """Advance the task and return the next trajectory command.
110
111        Args:
112            ally_zone (AllyZone): The ally zone information.
113            enemy_zone (EnemyZone): The enemy zone information.
114
115        Returns:
116            TrajectoryPlanCommand: Next trajectory command.
117        """
118        if self.state == NavigatorTaskState.NOT_PLANNED:
119            self._plan_task(ally_zone)
120            return self.trajectory_planner.get_plan()
121
122        if self._has_timed_out():
123            return self._abort()
124
125        cmd = self.trajectory_planner.get_plan()
126
127        if self._is_finished() and self.state != NavigatorTaskState.STABILIZING:
128            if self.params.stabilization_delay > 0:
129                self._stabilization_start_time = time.time()
130                self.state = NavigatorTaskState.STABILIZING
131            else:
132                self.state = NavigatorTaskState.FINISHED
133
134        if self.state == NavigatorTaskState.STABILIZING:
135            if self._get_stabilization_elapsed() < self.params.stabilization_delay:
136                return cmd
137            self.state = NavigatorTaskState.FINISHED
138
139        if self.state == NavigatorTaskState.FINISHED:
140            return cmd
141
142        avoidance_cmd = self.avoidance.handle(
143            current_navigator_task=self,
144            ally_zone=ally_zone,
145            enemy_zone=enemy_zone,
146        )
147        if self.state == NavigatorTaskState.AVOIDING:
148            return avoidance_cmd
149
150        self.current_trajectory_command = cmd
151        return self.current_trajectory_command

119            self._plan_task(ally_zone)
120            return self.trajectory_planner.get_plan()
121
122        if self._has_timed_out():
123            return self._abort()
124
125        cmd = self.trajectory_planner.get_plan()
126
127        if self._is_finished() and self.state != NavigatorTaskState.STABILIZING:
128            if self.params.stabilization_delay > 0:
129                self._stabilization_start_time = time.time()
130                self.state = NavigatorTaskState.STABILIZING
131            else:
132                self.state = NavigatorTaskState.FINISHED
133
134        if self.state == NavigatorTaskState.STABILIZING:
135            if self._get_stabilization_elapsed() < self.params.stabilization_delay:
136                return cmd
137            self.state = NavigatorTaskState.FINISHED
138
139        if self.state == NavigatorTaskState.FINISHED:
140            return cmd
141
142        avoidance_cmd = self.avoidance.handle(
143            current_navigator_task=self,
144            ally_zone=ally_zone,
145            enemy_zone=enemy_zone,
146        )
147        if self.state == NavigatorTaskState.AVOIDING:
148            return avoidance_cmd
149
150        self.current_trajectory_command = cmd
151        return self.current_trajectory_command

 66    def handle(
 67        self,
 68        current_navigator_task: NavigatorTask,
 69        ally_zone: AllyZone,
 70        enemy_zone: EnemyZone,
 71    ) -> TrajectoryPlanCommand:
 72        """Process avoidance logic based on current zones and navigation state.
 73
 74        Args:
 75            current_navigator_task (NavigatorTask):
 76                The current navigation task instance.
 77            ally_zone (AllyZone): Ally zone providing positional data.
 78            enemy_zone (EnemyZone): Enemy zone used for obstacle detection.
 79
 80        Returns:
 81            TrajectoryPlanCommand:
 82            The trajectory command after processing avoidance logic.
 83        """
 84        position: OrientedPoint = ally_zone.point
 85        self.logger.debug(
 86            f"Handling avoidance at position: {position}, current state: {self.state}",
 87        )
 88
 89        # 1. Timeout check
 90        if self._has_timed_out():
 91            self.logger.warning("Avoidance timed out. Aborting task.")
 92            return self._abort(current_navigator_task, position)
 93
 94        # 2. Obstacle detected → init backward task
 95        if (
 96            self.acs_detector.is_acs_triggered(ally_zone, enemy_zone)
 97            and self.state == AvoidanceState.IDLE
 98        ):
 99            self.logger.info(
100                "Obstacle detected. starting backward avoidance. "
101                f"Distance: {ally_zone.point.distance(enemy_zone.point)}",
102            )
103
104            # Create backward navigator task
105            self.backward_navigator_task = NavigatorTask(
106                NavigatorTaskParams(
107                    goal=None,
108                    timeout=None,
109                    stabilization_delay=0.0,
110                    path_planner_params=DeltaPathPlannerParams(
111                        distance=self.params.backward_distance,
112                    ),
113                    trajectory_planner_params=SequentialTrajectoryPlannerParams(
114                        direction=Direction.BACKWARD,
115                    ),
116                    speed_profiler=self.params.backward_speed_profiler,
117                    avoidance_params=NoAvoidanceParams(),
118                    acs_detection_profile_params=NoAcsDetectionProfileParams(),
119                ),
120            )
121
122            self.state = AvoidanceState.AVOIDING
123            current_navigator_task.state = NavigatorTaskState.AVOIDING
124            self._start_timer()
125            self.logger.debug(f"Timer started at: {self._avoiding_start_time}")
126
127            # Execute first backward command immediately
128            cmd = self.backward_navigator_task.handle(ally_zone, enemy_zone)
129            current_navigator_task.current_trajectory_command = cmd
130            return cmd
131
132        # 3. While avoiding → keep executing backward
133        if (
134            self.state == AvoidanceState.AVOIDING
135            and self.backward_navigator_task is not None
136        ):
137            self.logger.info("Executing backward avoidance maneuver.")
138            cmd: TrajectoryPlanCommand = self.backward_navigator_task.handle(
139                ally_zone,
140                enemy_zone,
141            )
142
143            # Clear the backward task if it has finished
144            if self.backward_navigator_task.state == NavigatorTaskState.FINISHED:
145                self.backward_navigator_task = None
146
147            current_navigator_task.current_trajectory_command = cmd
148            return cmd  # Return the command from the backward avoidance task
149
150        # 4. Finished backward, obstacle clear → replan
151        if (
152            self.state == AvoidanceState.AVOIDING
153            and self.backward_navigator_task is None
154        ):
155            self.logger.info("Obstacle cleared. Replanning trajectory.")
156
157            # Obstacle is no longer detected, replan from current position
158            last_params = cast(
159                "BasePathPlannerPlanPathParams",
160                current_navigator_task.path_planner.last_plan_path_params,
161            )
162            last_params.start = position  # Update start position to current location
163
164            self.logger.debug(f"Replanning from updated start: {position}")
165
166            new_path = current_navigator_task.path_planner.plan_path(last_params)
167            current_navigator_task.trajectory_planner.plan_trajectory(new_path)
168            current_navigator_task.trajectory_planner.start_planning()
169
170            self.logger.debug("Trajectory planner reset internal clock.")
171            # reset timer just for logging/manure measurement
172            self._reset_timer()
173            self.logger.debug("Timer reset after avoidance completion.")
174
175            self.state = AvoidanceState.IDLE
176            current_navigator_task.state = NavigatorTaskState.IN_PROGRESS
177
178            self.logger.info("Avoidance complete. Resuming normal operation.")
179            return cast(
180                "TrajectoryPlanCommand",
181                current_navigator_task.current_trajectory_command,
182            )  # Avoidance complete, continue as normal
183
184        # 4. Continue with current command
185        self.logger.debug(
186            "No avoidance action required. Continuing original trajectory.",
187        )
188        return cast(
189            "TrajectoryPlanCommand",
190            current_navigator_task.current_trajectory_command,
191        )

 37        """Handle the stop-and-wait avoidance logic.
 38

 74            self._start_timer()
 75            self.logger.debug(f"Timer started at: {self._avoiding_start_time}")

 80            self.state == AvoidanceState.AVOIDING

 84

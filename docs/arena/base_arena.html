206    def _pol_to_abs_cart(self, polars: np.ndarray) -> MultiPoint:
207        """Converts polar coordinates to absolute Cartesian coordinates.
208
209        Args:
210            polars (np.ndarray):
211                Array of polar coordinates in the form of (angle, distance).
212
213        Returns:
214            MultiPoint: Array of absolute Cartesian coordinates.
215        """
216        return MultiPoint(
217            [
218                (
219                    self.ally_zone.point.x
220                    + np.cos(self.ally_zone.point.theta - polars[i, 0]) * polars[i, 1],
221                    self.ally_zone.point.y
222                    + np.sin(self.ally_zone.point.theta - polars[i, 0]) * polars[i, 1],
223                )
224                for i in range(len(polars))
225            ],
226        )

250        self,
251        ally_position: OrientedPoint,
252        lidar_scan_polars: np.ndarray,  # Polars coordinates issued from the lidar scan
253        *,
254        optimized_update: bool = True,
255        _enemy_position: Point | None = None,  # Only for testing and simulation purpose
256    ) -> None:
257        """Updates the state of the arena, zones, and grid.
258
259        Args:
260            ally_position (OrientedPoint): Current position of the ally robot.
261            lidar_scan_polars (np.ndarray): Lidar scan data in polar coordinates.
262            optimized_update (bool, optional):
263                If ``True``, only updates intersecting zones. Defaults to ``True``.
264            _enemy_position (Point | None, optional):
265                Pre-defined enemy position. Defaults to None.
266        """
267        # 1.Compute enemy position if not directly provided in absolute coordinates
268        if not _enemy_position:
269            # Compute enemy position based on lidar scans -> match situation
270            enemy_position = self.compute_enemy_position(
271                lidar_scan_polars,
272                ally_position,
273            )
274        else:
275            # Use the provided enemy position -> testing or simulation
276            enemy_position = _enemy_position
277
278        # 2.Update ally and enemy zones (specific zone, there are not in self.zones)
279        self.ally_zone.update(self.team_color, ally_position, enemy_position)
280        self.enemy_zone.update(self.team_color, ally_position, enemy_position)
281
282        # 3.Update all other zones
283        # optimized: Update only the zones that intersect with the points
284        all_points: list[OrientedPoint | Point] = [ally_position, enemy_position]
285        for zone in self.zones:
286            if not optimized_update or any(
287                zone.polygon.contains(pt) for pt in all_points
288            ):
289                zone.update(
290                    self.team_color,
291                    ally_position=ally_position,
292                    enemy_position=enemy_position,
293                )

298    def compute_enemy_position(
299        self,
300        lidar_scan_polars: np.ndarray,
301        ally_position: OrientedPoint,
302        *,
303        _start_time: int = -1,
304        _numb_enemy: bool = False,
305    ) -> Point | OrientedPoint:
306        """Computes the position of the enemy and updates the arena.
307
308        This function calculates the position of the enemy by processing the
309        lidar scans.
310        It removes any obstacles that are outside the arena, and then determines
311        the closest obstacle as the enemy position.
312        If the enemy position is within a stuff zone, it marks that zone as FORBIDDEN.
313
314        Args:
315            lidar_scan_polars (np.ndarray): Detection points from the LIDAR scan.
316            ally_position (OrientedPoint): Current ally position.
317            _start_time (int, optional):
318                Starting timestamp for the computation. Defaults to -1.
319            _numb_enemy (bool, optional):
320                Whether the enemy is inactive. Defaults to ``False``.
321
322        Returns:
323            Point | OrientedPoint: The computed enemy position.
324        """
325        obstacles: MultiPoint = self.remove_outside(
326            self._pol_to_abs_cart(lidar_scan_polars),
327        )
328
329        if not is_empty(obstacles):
330            return nearest_points(ally_position, obstacles)[1]
331
332        return self.enemy_zone.point

347    def compute_goal_position(
348        self,
349        goal: int | BaseArenaZone | OrientedPoint | Point,
350    ) -> OrientedPoint | Point | None:
351        """Compute a goal position based on zone or point information.
352
353        Args:
354            goal (int | BaseArenaZone | OrientedPoint | Point): Zone identifier
355                or direct destination.
356
357        Returns:
358            OrientedPoint | Point | None: Computed destination or ``None`` if the
359            goal type is invalid.
360        """
361        # 1. If goal is defined as int, it's a zone ID
362        if isinstance(goal, int):
363            if goal > len(self.zones):
364                self.logger.error("Invalid zone ID given in trajectory parameters.")
365                return None
366
367            return self.zones[goal].get_go_to_position(
368                ally_position=self.ally_zone.point,
369                team_color=self.team_color,
370            )
371
372        # 2. If goal is a BaseArenaZone -> compute the best goal point
373        if isinstance(goal, BaseArenaZone):
374            return goal.get_go_to_position(
375                ally_position=self.ally_zone.point,
376                team_color=self.team_color,
377            )
378
379        # 3. If goal is an OrientedPoint or Point, return it as is
380        return goal

591            label = None
592

596            ax.fill(
597                x,
598                y,
599                alpha=alpha,
600                fc=color,
601                label=label,
602                hatch=hatch,
603                ec=hatch_color or color,
604            )
605        else:
606            # Polygon with holes: outline and interior lines
607            x, y = polygon.exterior.xy
608            ax.plot(x, y, color=color, linestyle="--", label=label, alpha=alpha)
609            for interior in polygon.interiors:
610                x, y = interior.xy
611                ax.plot(x, y, color=color, linestyle="--", alpha=alpha)

613    def __plot_zone(
614        self,
615        ax: plt.Axes,
616        zone: BaseArenaZone,
617        *,
618        show_buffer: bool,
619        show_ally_direction: bool,
620        display_zones_go_to_positions: bool,
621        transparency_factor: float = 1.0,
622    ) -> None:
623        """Plot a zone and its buffer on the arena.
624
625        Args:
626            ax (plt.Axes): Axis on which to draw.
627            zone (BaseArenaZone): Zone to display.
628            show_buffer (bool): Whether to plot the buffered polygon.
629            show_ally_direction (bool): Draw an arrow for the ally direction.
630            display_zones_go_to_positions (bool): Draw go-to positions if any.
631            transparency_factor (float, optional):
632                Alpha value multiplier. Defaults to 1.0.
633        """
634        if show_buffer:
635            # Plot buffer zone in transparent color
636            self.__plot_polygon(
637                ax,
638                zone.buffered_polygon,
639                color=zone.zone_color,
640                alpha=0.5 * transparency_factor,
641            )
642
643        if isinstance(zone, BorderZone):
644            self.__plot_polygon(
645                ax,
646                zone.polygon,
647                color=zone.zone_color,
648                alpha=0.5 * transparency_factor,
649            )
650        else:
651            # 1. Define hatch parameters (depends on zone accessibility)
652            hatch_params = self.__get_hatch_parameters(zone)
653
654            # 2. Plot zone uid
655            # Don't plot uid for ally and enemy zones (they increase at each update)
656            if not isinstance(zone, AllyZone) and not isinstance(zone, EnemyZone):
657                self.__plot_zone_uid(ax, zone)
658
659            # 3. Plot the zone polygon
660            self.__plot_polygon(
661                ax,
662                zone.polygon,
663                color=zone.zone_color,
664                label=zone.zone_type.name,
665                alpha=0.8 * transparency_factor,
666                **hatch_params,
667            )
668
669            # 4. Plot go-to positions
670            if display_zones_go_to_positions and zone.go_to_positions:
671                # Get the go-to position nearest to the ally robot

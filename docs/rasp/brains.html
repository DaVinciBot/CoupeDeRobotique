 99    def run(self) -> None:
100        """Runs the main control loop for the robot."""
101        # --- Initialization --- #
102        # --- 1) Initialize subsystems --- #
103        rolling_basis: RollingBasis | RollingBasisDummy = RollingBasisDummy(
104            logger=Logger(identifier="RollingBasis", follow_logger_manager_rules=True),
105        )
106        rolling_basis.set_odometrie(self.rolling_basis_odometrie)
107        rolling_basis.initialize_pids()
108
109        actuators: ActuatorsShow | ActuatorsShowDummy = ActuatorsShowDummy(
110            logger=Logger(identifier="Actuators", follow_logger_manager_rules=True),
111        )
112        actuators.deplacement_position()
113        # --- 2) Wait for jack plug ● Deploy banner block ● Wait for trigger --- #
114        while not self.jack_plugged:  # wait until cable is plugged
115            time.sleep(0.1)
116        actuators.block_banner()  # engage the banner blocker
117        rolling_basis.set_odometrie(self.rolling_basis_odometrie)
118        rolling_basis.initialize_pids()
119        while not self.jack_triggered:  # wait for the trigger event
120            time.sleep(0.1)
121
122        # --- 3) Build the strategy --- #
123
124        strategy = TowerRushAltStrategy(
125            ShowGameContext(
126                arena=self.arena,
127                rolling_basis=rolling_basis,
128                actuators=actuators,
129                score=self.score,
130            ),
131        )
132
133        # from strategy.tools import visualize_task_graph
134        # visualize_task_graph(strategy.runner.active[0])
135
136        # --- MetaProg is insane (loop) --- #
137        context = ShowGameContext(
138            arena=self.arena,
139            rolling_basis=rolling_basis,
140            actuators=actuators,
141            score=self.score,
142        )
143
144        strategy.runner.handle(context)
145
146        # Update the rolling basis odometrie from the context
147        self.score = context.score
148        self.ui_state["score"] = self.score
149        self.rolling_basis_odometrie = rolling_basis.odometrie
150        self.ui_state["odometrie_state"] = rolling_basis.odometrie

217    @Brain.task(process=False, run_on_start=True, refresh_rate=0.5)
218    async def receive_ui_data(self) -> None:
219        """Executes requests received by the server.
220
221        Use Postman to send request to the server
222        Use eval and await eval to run the code you want. Code must be sent as a string
223        """
224        ui = await self.ws_ui.receiver.get()
225
226        if ui != WSmsg():
227            self.logger.info(f"UI instruction {ui.msg} received: {ui.data}")
228            if ui.msg == "eval":
229                instructions = []
230                if isinstance(ui.data, str):
231                    instructions.append(ui.data)
232                elif isinstance(ui.data, list):
233                    instructions = ui.data
234
235                for instruction in instructions:
236                    if instruction.startswith("await "):
237                        await ast.literal_eval(instruction.removeprefix("await "))
238                    else:
239                        ast.literal_eval(instruction)
240            elif ui.msg == "team change":
241                if ui.data["team"] in {"yellow", "blue"}:
242                    self.arena.set_team_color(TeamColor[ui.data["team"].upper()])
243                    self.logger.info(f"Team color set to {ui.data['team']}")
244                else:
245                    self.logger.warning(f"Invalid team color: {ui.data}")
246            else:
247                self.logger.warning(f"Command not implemented: {ui.msg} / {ui.data}")

174    def run(self) -> None:
175        """Runs the main control loop for the robot."""
176        # --- Initialization --- #
177        strategy: GoBackstageStrategy | None = None
178        action_holder: list[GraphRunner | None] = [None]
179        init_stage = "rolling basis setup"
180        try:
181            # --- 1) Initialize subsystems --- #
182            if CONFIG.ROLLING_BASIS_DUMMY:
183                rolling_basis: RollingBasis | RollingBasisDummy = RollingBasisDummy(
184                    logger=LogLogger(
185                        identifier="RollingBasisDummy",
186                        follow_logger_manager_rules=True,
187                    ),
188                    enable_realtime_simulation=True,
189                    enable_debug_report=True,
190                )
191            else:
192                rolling_basis = RollingBasis(
193                    logger=LogLogger(
194                        identifier="RollingBasis",
195                        follow_logger_manager_rules=True,
196                    ),
197                )
198
199            init_stage = "rolling basis PID disable before jack plug"
200            rolling_basis.set_odometrie(self.rolling_basis_odometrie)
201            set_zero_pids(rolling_basis)
202
203            init_stage = "actuators setup"
204            if CONFIG.ACTUATORS_DUMMY:
205                actuators: ActuatorsShow | ActuatorsShowDummy = ActuatorsShowDummy(
206                    logger=LogLogger(
207                        identifier="Actuators",
208                        follow_logger_manager_rules=True,
209                    ),
210                )
211            else:
212                actuators = ActuatorsShow(
213                    logger=LogLogger(
214                        identifier="Actuators",
215                        follow_logger_manager_rules=True,
216                    ),
217                )
218
219            # --- 2) Wait for jack plug ● Deploy banner block ● Wait for trigger --- #
220            init_stage = "jack plug wait"
221            if (
222                not CONFIG.LIDAR_DUMMY
223                or not CONFIG.ROLLING_BASIS_DUMMY
224                or not CONFIG.ACTUATORS_DUMMY
225            ):
226                while not self.jack_plugged:  # wait until cable is plugged
227                    time.sleep(0.1)
228            else:
229                time.sleep(2)
230
231            init_stage = "rolling basis reinitialization"
232            rolling_basis.set_odometrie(self.rolling_basis_odometrie)
233            rolling_basis.initialize_pids()
234
235            init_stage = "jack trigger wait"
236            while not self.jack_triggered:  # wait for the trigger event
237                time.sleep(0.1)
238
239            match_start_time = time.monotonic()
240            normal_match_timeout_sent = False
241
242            # --- 3) Build the strategy --- #
243            init_stage = "strategy creation"
244            if self.mode == "iihm":
245                self.logger.info("IIHM mode: Waiting for first task...")
246            else:
247                strategy = GoBackstageStrategy(
248                    WinterGameContext(
249                        arena=self.arena,
250                        rolling_basis=rolling_basis,
251                        actuators=actuators,
252                        point=self.score,
253                    ),
254                    rotation_sign=get_relative_rotation_sign(self.arena.team_color),
255                )
256
257            self.should_send_start = True
258            self.status = "starting"
259        except Exception:
260            self.logger.error(
261                f"[run] Initialization failed during {init_stage}: "
262                f"{traceback.format_exc()}",
263            )
264            raise
265
266        # from strategy.tools import visualize_task_graph
267        # visualize_task_graph(strategy.runner.active[0])
268
269        # --- MetaProg is insane (loop) --- #
270        normal_match_timed_out = (
271            self.mode == "normal"
272            and time.monotonic() - match_start_time >= NORMAL_MATCH_TIMEOUT_S
273        )
274
275        context = WinterGameContext(
276            arena=self.arena,
277            rolling_basis=rolling_basis,
278            actuators=actuators,
279            point=self.score,
280        )
281
282        if normal_match_timed_out:
283            action_holder[0] = None
284            if not normal_match_timeout_sent:
285                self.logger.info(
286                    "[BRAIN:Match] Normal mode timeout reached; stopping robot.",
287                )
288                rolling_basis.set_target_position(rolling_basis.odometrie)
289                rolling_basis.set_motors_pwm(
290                    left_pwm=0,
291                    right_pwm=0,
292                    duration_ms=100,
293                )
294                normal_match_timeout_sent = True
295        elif strategy:
296            strategy.runner.handle(context)
297        else:
298            if self.should_update_task:
299                zone: int | None = self.arena.get_current_zone_id()
300                if self.task_type == "navigation":
301                    action_holder[0] = GraphRunner(
302                        logger=LogLogger(
303                            identifier="IIHMRunner",
304                            follow_logger_manager_rules=True,
305                        ),
306                        start=BaseTaskNode(
307                            name=f"[Debug] {self.task_name}",
308                            tasks=self.task_todo,
309                        ),
310                    )
311                elif self.task_type == "relative_forward":
312                    distance = float(self.task_data.get("distance", 0.0))
313                    action_holder[0] = GraphRunner(
314                        logger=LogLogger(
315                            identifier="IIHMRunner",
316                            follow_logger_manager_rules=True,
317                        ),
318                        start=BaseTaskNode(
319                            name=f"[Debug] Relative forward {distance} cm",
320                            tasks=[RelativeForward(distance)],
321                        ),
322                    )
323                elif self.task_type == "relative_backward":
324                    distance = float(self.task_data.get("distance", 0.0))
325                    action_holder[0] = GraphRunner(
326                        logger=LogLogger(
327                            identifier="IIHMRunner",
328                            follow_logger_manager_rules=True,
329                        ),
330                        start=BaseTaskNode(
331                            name=f"[Debug] Relative backward {distance} cm",
332                            tasks=[RelativeBackward(distance)],
333                        ),
334                    )
335                elif self.task_type == "relative_turn":
336                    angle = float(self.task_data.get("angle", 0.0))
337                    action_holder[0] = GraphRunner(
338                        logger=LogLogger(
339                            identifier="IIHMRunner",
340                            follow_logger_manager_rules=True,
341                        ),
342                        start=BaseTaskNode(
343                            name=f"[Debug] Relative turn {angle} rad",
344                            tasks=[RelativeRotation(angle)],
345                        ),
346                    )
347                elif self.task_type == "banner_deploy":
348                    action_holder[0] = GraphRunner(
349                        logger=Logger(
350                            identifier="IIHMRunner",
351                            follow_logger_manager_rules=True,
352                        ),
353                        start=get_banner_deployment_subgraph().get_entry(),
354                    )
355                elif self.task_type == "construct" and zone is not None:
356                    action_holder[0] = GraphRunner(
357                        logger=Logger(
358                            identifier="IIHMRunner",
359                            follow_logger_manager_rules=True,
360                        ),
361                        start=get_construct_subgraph(
362                            zone,
363                        ).get_entry(),
364                    )
365                elif self.task_type == "pickup" and zone is not None:
366                    action_holder[0] = GraphRunner(
367                        logger=Logger(
368                            identifier="IIHMRunner",
369                            follow_logger_manager_rules=True,
370                        ),
371                        start=get_pickup_subgraph(
372                            zone,
373                        ).get_entry(),
374                    )
375                elif self.task_type == "push_floor" and zone is not None:
376                    action_holder[0] = GraphRunner(
377                        logger=Logger(
378                            identifier="IIHMRunner",
379                            follow_logger_manager_rules=True,
380                        ),
381                        start=get_push_one_floor_to_wall_subgraph(
382                            zone,
383                            30,
384                        ).get_entry(),
385                    )
386                elif self.task_type == "construct_one_floor" and zone is not None:
387                    action_holder[0] = GraphRunner(
388                        logger=Logger(
389                            identifier="IIHMRunner",
390                            follow_logger_manager_rules=True,
391                        ),
392                        start=get_construct_one_floor_subgraph(
393                            zone,
394                        ).get_entry(),
395                    )
396                elif self.task_type == "reset_odometry":
397                    action_holder[0] = GraphRunner(
398                        logger=Logger(
399                            identifier="IIHMRunner",
400                            follow_logger_manager_rules=True,
401                        ),
402                        start=BaseTaskNode(
403                            name="[Debug] Reset Odometry",
404                            tasks=[
405                                SetOdometrie(0, 0, 0),
406                            ],
407                        ),
408                    )
409                self.should_update_task = False
410            if action_holder[0] is not None:
411                action_holder[0].handle(context)
412
413        if self.should_send_direct_pwm and not normal_match_timed_out:
414            action_holder[0] = None
415            rolling_basis.set_motors_pwm(
416                left_pwm=self.direct_pwm_left,
417                right_pwm=self.direct_pwm_right,
418                duration_ms=self.direct_pwm_duration_ms,
419            )
420            self.should_send_direct_pwm = False
421        elif self.should_send_direct_pwm and normal_match_timed_out:
422            self.logger.warning(
423                "[BRAIN:Match] Ignoring direct PWM request after normal mode timeout.",
424            )
425            self.should_send_direct_pwm = False
426
427        if self.should_update_pid:
428            if self.pid_type in {"linear", "linear_position"}:
429                rolling_basis.set_linear_position_pid(
430                    kp=self.pid_kp,
431                    ki=self.pid_ki,
432                    kd=self.pid_kd,
433                )
434            elif self.pid_type in {"angular", "angular_position"}:
435                rolling_basis.set_angular_position_pid(
436                    kp=self.pid_kp,
437                    ki=self.pid_ki,
438                    kd=self.pid_kd,
439                )
440            elif self.pid_type in {"left_wheel", "left_wheel_position"}:
441                rolling_basis.set_left_wheel_position_pid(
442                    kp=self.pid_kp,
443                    ki=self.pid_ki,
444                    kd=self.pid_kd,
445                )

447                rolling_basis.set_right_wheel_position_pid(
448                    kp=self.pid_kp,

451                )
452            self.should_update_pid = False
453
454        # Update shared state from the context
455        self.score = context.point
456        self.odemetrie_state = context.arena.ally_zone.point
457        self.enemy_odemetrie_state = context.arena.enemy_zone.point
458        self.rolling_basis_odometrie = rolling_basis.odometrie
459        self.pid_debug_live = rolling_basis.get_debug_snapshot()
460
461        if self.should_export_debug_report:
462            rolling_basis.export_debug_report(reason="shutdown_request")
463            self.should_export_debug_report = False

550    @Brain.task(process=False, run_on_start=True, refresh_rate=0.5)
551    async def receive_ui_data(self) -> None:
552        """Executes requests received by the server.
553
554        Use Postman to send request to the server
555        Use eval and await eval to run the code you want. Code must be sent as a string
556        """
557        ui = await self.ws_ui.receiver.get()
558
559        if ui != WSmsg():
560            self.logger.info(f"[WS:UI] Instruction {ui.msg} | data: {ui.data}")
561            if ui.msg == "team change":
562                if ui.data["team"] in {"yellow", "blue"}:
563                    self.arena.set_team_color(TeamColor[ui.data["team"].upper()])
564                    self.logger.info(f"Team color set to {ui.data['team']}")
565                else:
566                    self.logger.warning(f"Invalid team color: {ui.data}")
567            elif ui.msg == "mode change":
568                if ui.data["mode"] in {"normal", "iihm"}:
569                    self.mode = ui.data["mode"]
570                    self.logger.info(f"Mode set to {ui.data['mode']}")
571                    await self.ws_ui.sender.send(
572                        WSmsg(
573                            sender="server",
574                            msg="mode set",
575                            data={"mode": self.mode},
576                        ),
577                    )
578                else:
579                    self.logger.warning(f"Invalid mode: {ui.data}")
580            elif ui.msg == "hello":
581                await self.receive_hello()
582            elif ui.msg == "action":
583                if ui.data["type"] == "go to point":
584                    x = float(ui.data["data"]["x"])
585                    y = float(ui.data["data"]["y"])
586                    theta = float(ui.data["data"]["theta"])
587                    self.task_name = f"Go to point ({x}, {y}, {theta})"
588                    self.task_todo = [
589                        GoToOrientedPoint(OrientedPoint(x, y, theta)),
590                    ]
591                    self.task_type = "navigation"
592                    self.should_update_task = True
593                    self.logger.info(
594                        f"Set task to {self.task_name}",
595                    )
596                elif ui.data["type"] in {
597                    "relative_forward",
598                    "relative_backward",
599                    "relative_turn",
600                }:
601                    self.task_type = str(ui.data["type"])
602                    self.task_data = dict(ui.data.get("data", {}))
603                    self.task_name = self.task_type
604                    self.should_update_task = True
605                    self.logger.info(
606                        f"Set task to {self.task_type}: {self.task_data}",
607                    )
608                elif ui.data["type"] == "direct_pwm":
609                    self.direct_pwm_left = int(ui.data["data"].get("left_pwm", 0))
610                    self.direct_pwm_right = int(ui.data["data"].get("right_pwm", 0))
611                    self.direct_pwm_duration_ms = int(
612                        ui.data["data"].get("duration_ms", 500),
613                    )
614                    self.should_send_direct_pwm = True
615                    self.logger.info(
616                        "Direct PWM request: "
617                        f"left={self.direct_pwm_left}, "
618                        f"right={self.direct_pwm_right}, "
619                        f"duration={self.direct_pwm_duration_ms}ms",
620                    )
621                elif ui.data["type"] in {
622                    "banner_deploy",
623                    "construct",
624                    "pickup",
625                    "push_floor",
626                    "construct_one_floor",
627                    "reset_odometry",
628                }:
629                    self.task_type = ui.data["type"]
630                    self.should_update_task = True
631                    self.logger.info(
632                        f"Set task to {self.task_name}",
633                    )
634                else:
635                    self.logger.warning(f"Unknown action type: {ui.data['type']}")
636            elif ui.msg == "pid update":
637                kp = float(ui.data["data"]["kp"])
638                ki = float(ui.data["data"]["ki"])
639                kd = float(ui.data["data"]["kd"])
640                pid_type: str = str(ui.data["type"])
641                self.pid_kp = kp
642                self.pid_ki = ki
643                self.pid_kd = kd
644                self.logger.info(
645                    f"Updating {pid_type} PID to Kp: {kp}, Ki: {ki}, Kd: {kd}",
646                )
647                self.pid_type = pid_type
648                self.should_update_pid = True
649            else:
650                self.logger.warning(
651                    f"[WS:UI] Command not implemented: {ui.msg} / {ui.data}",
652                )

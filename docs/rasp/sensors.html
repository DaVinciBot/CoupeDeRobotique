 28
 29    * The angles are in degrees (considering the sense in the trigonometric way).

 34    * the unities of angles and distances can be changed by changing the default values

 86    def __init_lidar(self) -> LidarT:
 87        """Initialize the lidar object and test the connection.
 88
 89        Returns:
 90            LidarT: the lidar object
 91
 92        Raises:
 93            ConnectionError: If the lidar is not connected or does not work correctly.
 94            ImportError: If the lidar module cannot be imported.
 95        """
 96        try:
 97            import pysicktim as lidar  # noqa: PLC0415
 98
 99            lidar = reload(lidar)
100        except Exception as error:
101            self._logger.critical(
102                f"[SENSOR:Lidar:Init] Failed to import pysicktim: {error},",
103            )
104            msg = f"Error while importing lidar [{error}]!"
105            raise ImportError(msg) from error
106
107        if getattr(lidar, "lidar", None) is None:
108            self._logger.warning("[SENSOR:Lidar:Init] USB device not found")
109            msg = "Lidar USB device not found !"
110            raise ConnectionError(msg)
111
112        # Test lidar connection by testing scan function
113        try:
114            lidar.scan()
115        except Exception as error:
116            self._logger.warning(
117                f"[SENSOR:Lidar:Init] Scan test failed: {error}",
118            )
119            msg = f"Lidar doesn't work correctly [{error}]!"
120            raise ConnectionError(msg) from error
121
122        if lidar.scan.distances is None or lidar.scan.distances == []:
123            self._logger.critical("[SENSOR:Lidar:Init] Scan test failed - no data")
124            msg = "Lidar doesn't work correctly !"
125            raise ConnectionError(msg)
126
127        self._logger.info("[SENSOR:Lidar:Init] Connected successfully")
128        return lidar

130    def __threading_init_lidar(self) -> None:
131        """Initialize the lidar in a thread.
132
133        It will retry to initialize the lidar until it is connected.
134        """
135
136        def init() -> None:
137            while not self.__is_connected:
138                try:
139                    self._logger.debug(
140                        "[SENSOR:Lidar:Init] Try to initialize lidar in thread ...",
141                    )
142                    self.__lidar_obj = self.__init_lidar()
143                    # Initialize polars based on the lidar measurement count
144                    self.__polars_angles = self.__init_polars_angle(
145                        self.__min_angle,
146                        self.__max_angle,
147                    )
148                    self.__is_connected = True
149                except (ConnectionError, ImportError, ValueError) as error:
150                    self._logger.warning(
151                        f"[SENSOR:Lidar:Init] Initialization failed: {error} "
152                        f"- retrying in {self.__initialization_fail_refresh_rate}s",
153                    )
154                    time.sleep(self.__initialization_fail_refresh_rate)
155
156        thread = threading.Thread(target=init)
157        thread.start()

159    def __init_polars_angle(
160        self,
161        min_angle: float,
162        max_angle: float,
163    ) -> NDArray[np.float32]:
164        """Initialize the polar angles array.
165
166        Args:
167            min_angle (float): Minimum angle of the lidar (max angle at left).
168            max_angle (float): Maximum angle of the lidar (min angle at right).
169
170        Returns:
171            NDArray[np.float32]: numpy array of angles centered around 0°.
172
173        Raises:
174            ValueError: If the polars array cannot be initialized.
175        """
176        n = len(self.distances)  # Number of distances
177        if not n:
178            self.__scan()
179            n = len(self.distances)  # Number of distances
180
181        angle_step = abs(max_angle - min_angle) / n
182
183        # Init the polars array with zeros.
184        # Then fill it with angles centered, so that the "front" of the lidar is 0
185        centered_polars = np.zeros(n, dtype=np.float32)
186        for i in range(n):
187            centered_polars[i] = -((max_angle - min_angle) / 2) + i * angle_step
188
189        if not centered_polars.size:
190            msg = "[SENSOR:Lidar:Init] Error while initializing polars"
191            self._logger.critical(msg)
192            raise ValueError(msg)
193
194        return centered_polars
195
196    def __init_angles_unit(self, unit: str) -> float:
197        """Initialize the unit of the angles.
198
199        Args:
200            unit (str): The unit of the angles.
201
202        Returns:
203            float: conversion factor for the angles
204
205        Raises:
206            ValueError: If the unit is not recognized.
207        """
208        if unit == "deg":
209            return 1
210        if unit == "rad":
211            return math.pi / 180
212
213        msg = f"[SENSOR:Lidar:Init] Unit of angles not recognized: {unit}"
214        self._logger.critical(msg)
215        raise ValueError(msg)
216
217    def __init_distances_unit(self, unit: str) -> float:
218        """Initialize the unit of the distances.
219
220        Args:
221            unit (str): The unit of the distances.
222
223        Returns:
224            float: conversion factor for the distances
225
226        Raises:
227            ValueError: If the unit is not recognized.
228        """
229        if unit == "mm":
230            return 1000
231        if unit == "cm":
232            return 100
233        if unit == "m":
234            return 1
235        if unit == "inch":
236            return 0.0254
237
238        msg = f"[SENSOR:Lidar:Init] Unit of distances not recognized: {unit}"
239        self._logger.critical(msg)
240        raise ValueError(msg)
241
242    def __scan(self) -> None:
243        """Scan the environment and store the distances.
244
245        If the scan fails, it will try to reconnect the lidar.
246
247        Raises:
248            LidarError: If the lidar is disconnected or if the scan fails.
249        """
250        if self.__lidar_obj is None:
251            msg = "LiDAR is not initialized yet"
252            raise LidarError(msg)
253
254        try:
255            self.__lidar_obj.scan()
256        except Exception as error:
257            # LiDAR seems to be disconnected
258            self._logger.error(
259                f"[SENSOR:Lidar] Scan failed - attempting reconnection: {error}",
260            )
261            # Try to reconnect LiDAR if it was connected before
262            if self.__is_connected:
263                self.__is_connected = False
264                self.__threading_init_lidar()
265
266            msg = f"Error while scanning, LiDAR is disconnected ? [{error}]"
267            raise LidarError(msg) from error

134    def scan_to_distances(self) -> NDArray[np.float32]:
135        """Simulate a lidar scan and return fake distance data.
136
137        Returns:
138            NDArray[np.float32]: numpy array of distances
139        """
140        # Initialize an array of distances
141        distances = np.full(
142            self.__num_points,
143            5.0,
144            dtype=np.float32,
145        )  # Default max range
146
147        # Simulate obstacles as clusters of points
148        num_obstacles = np.random.default_rng().integers(3, 8)  # Number of obstacles
149        for _ in range(num_obstacles):
150            center_angle = np.random.default_rng().uniform(
151                0,
152                self.__num_points,
153            )  # Random angle for obstacle
154            obstacle_width = np.random.default_rng().integers(

165            end_idx = int(min(self.__num_points, center_angle + obstacle_width // 2))
166            distances[start_idx:end_idx] = (
167                np.random
168                .default_rng()
169                .uniform(
170                    obstacle_distance - 0.1,
171                    obstacle_distance + 0.1,
172                    size=(end_idx - start_idx),
173                )
174                .astype(np.float32)
175            )
176
177        # Add noise for realism
178        distances += (
179            np.random
180            .default_rng()
181            .normal(0, 0.01, self.__num_points)
182            .astype(np.float32)
183        )
184        distances = np.clip(
185            distances,
186            0.1,
187            5.0,
188        )  # Ensure distances are within sensor range

191            "[SENSOR:Lidar:Dummy] Simulated realistic distances generated.",

 72    def _auto_build_transitions(
 73        self,
 74        *elements: BaseTaskNode | BaseSubGraph,
 75    ) -> bool:
 76        """Build a subgraph using the provided TaskNodes or SubGraphs.
 77
 78        Accepts any number of arguments of type BaseTaskNode or BaseSubGraph.
 79
 80        Args:
 81            *elements (BaseTaskNode | BaseSubGraph):
 82                A variable number of nodes or subgraphs.
 83
 84        Returns:
 85            bool:
 86                ``True`` if the transitions were built successfully,
 87                ``False`` otherwise.
 88        """
 89        if not elements:
 90            self._logger.error("[STRAT] No elements provided for building")
 91            return False
 92
 93        # Resolve entry and exit points for each element
 94        entry_points = [self._resolve_for_entry(el) for el in elements]
 95        exit_points = [self._resolve_for_exits(el) for el in elements]
 96        for i in range(len(entry_points) - 1):
 97            # Create a direct transition from the exit of the current element
 98            # to the entry of the next
 99            self._logger.debug(
100                f"[STRAT] Creating transition: "
101                f"{exit_points[i].name} -> {entry_points[i + 1].name}",
102            )
103            exit_points[i].add_transition(DirectTransition(entry_points[i + 1]))
104        return True
105
106    @staticmethod
107    def _resolve_for_entry(element: BaseTaskNode | BaseSubGraph) -> BaseTaskNode:
108        """Resolve a TaskNode or SubGraph to its entry point.
109
110        Args:
111            element (BaseTaskNode | BaseSubGraph): The element to resolve.
112
113        Returns:
114            BaseTaskNode: The entry point of the resolved element.
115        """
116        if isinstance(element, BaseSubGraph):
117            return element.get_entry()
118        return element
119
120    @staticmethod
121    def _resolve_for_exits(element: BaseTaskNode | BaseSubGraph) -> BaseTaskNode:
122        """Resolve a TaskNode or SubGraph to its exit points.
123
124        Args:
125            element (BaseTaskNode | BaseSubGraph): The element to resolve.
126
127        Returns:
128            BaseTaskNode: The exit point of the resolved element.
129        """
130        if isinstance(element, BaseSubGraph):
131            return element.get_exits()[0]
132        return element

148
149        Args:

152        super().__init__(
153            goal=zone_id,
154            path_planner_params=BasicPathPlannerParams(),
155            trajectory_planner_params=SequentialTrajectoryPlannerParams(
156                step_sleep_delay=2,
157            ),
158            speed_profiler=CONFIG.ROLLING_BASIS_SLOW_SPEED_PROFILER,
159            avoidance_params=StopAndWaitAvoidanceParams(timeout=20),
160            acs_detection_profile_params=RectangularProjectionAcsDetectionProfileParams(
161                acs_distance=20,
162                width_view=40,
163            ),
164            stabilization_delay=0.5,
165            points=0,
166        )

169        self.navigator_task: NavigatorTask
170        self.estimated_duration = self.compute_estimated_duration

172    def compute_estimated_duration(self, ctx: WinterGameContext) -> float:
173        """Estimate the duration to reach the centroid of the zone.
174
175        Args:
176            ctx (WinterGameContext): The game context providing arena information.
177
178        Returns:
179            float: Estimated duration in seconds.
180        """
181        centroid: OrientedPoint = OrientedPoint.from_point(
182            ctx.arena.zones[self.zone_id].polygon.centroid,
183        )
184        dist: float = float(distance(centroid, ctx.arena.ally_zone.point))
185        return (
186            self.speed_profiler.linear_speed_profile.get_total_duration(
187                distance=dist,
188            )
189            + self.stabilization_delay
190        )

206        # Create a NavigatorTask using the calculated goal

25class GoToStuffZoneToPickUp(NavigationTask):
26    """Navigate the robot to a specified stuff zone for a pickup.
27
28    Configures path planning, trajectory planning, speed profiling, and
29    avoidance strategies. Uses a rectangular projection ACS detection profile to
30    avoid collisions and applies a stabilization delay for system readiness
31    before performing tasks.
32    """
33
34    def __init__(self, stuff_zone_id: int, ctx: WinterGameContext) -> None:
35        """Initialize navigation and avoidance parameters.
36
37        Args:
38            stuff_zone_id (int): Identifier for the target stuff zone location.
39            ctx (WinterGameContext): The current game context.
40
41        Raises:
42            ValueError: If the provided color_reserved_zone_id is invalid.
43        """
44        goal = ctx.arena.compute_goal_position(stuff_zone_id)
45
46        if goal is None:
47            msg = f"Invalid stuff zone ID: {stuff_zone_id}"
48            raise ValueError(msg)
49
50        super().__init__(
51            goal=goal,
52            path_planner_params=AStarPathPlannerParams(
53                grid=ctx.arena.grid_manager.get_static_and_dynamic_grid(),
54                path_resolution=5,
55                chunk_size=CONFIG.ARENA_CHUNK_SIZE,
56                start=ctx.arena.ally_zone.point,
57                goal=goal,
58                direction=Direction.FORWARD,
59            ),
60            trajectory_planner_params=SequentialTrajectoryPlannerParams(
61                step_sleep_delay=2,
62            ),
63            # Use for pickup speed profiler
64            speed_profiler=CONFIG.ROLLING_BASIS_DEFAULT_SPEED_PROFILER,
65            avoidance_params=StopAndWaitAvoidanceParams(timeout=20),
66            acs_detection_profile_params=RectangularProjectionAcsDetectionProfileParams(
67                acs_distance=55,
68                width_view=40,
69            ),
70            stabilization_delay=2,
71            points=0,
72        )
73        self.zone_id: int = stuff_zone_id
74        self.estimated_duration = self.compute_estimated_duration
75
76    def compute_estimated_duration(self, ctx: WinterGameContext) -> float:
77        """Estimate duration based on distance to the target stuff zone.
78
79        Args:
80            ctx (WinterGameContext): The current game context.
81
82        Returns:
83            float: Estimated time to reach the stuff zone in seconds.
84        """
85        target_point: OrientedPoint = OrientedPoint.from_point(
86            ctx.arena.zones[self.zone_id].polygon.centroid,
87        )
88        current_position: OrientedPoint = ctx.rolling_basis.odometrie
89        dist: float = distance(current_position, target_point)
90
91        return (
92            self.speed_profiler.linear_speed_profile.get_total_duration(
93                distance=dist,
94            )

 73    def _auto_build_transitions(
 74        self,
 75        *elements: BaseTaskNode | BaseSubGraph,
 76    ) -> bool:
 77        """Build a subgraph using the provided TaskNodes or SubGraphs.
 78
 79        Accepts any number of arguments of type BaseTaskNode or BaseSubGraph.
 80
 81        Args:
 82            *elements (BaseTaskNode | BaseSubGraph):
 83                A variable number of nodes or subgraphs.
 84
 85        Returns:
 86            bool:
 87                ``True`` if the transitions were built successfully,
 88                ``False`` otherwise.
 89        """
 90        if not elements:
 91            self._logger.error("[STRAT] No elements provided for building")
 92            return False
 93
 94        # Resolve entry and exit points for each element
 95        entry_points = [self._resolve_for_entry(el) for el in elements]
 96        exit_points = [self._resolve_for_exits(el) for el in elements]
 97        for i in range(len(entry_points) - 1):
 98            # Create a direct transition from the exit of the current element
 99            # to the entry of the next
100            self._logger.debug(
101                f"[STRAT] Creating transition: "
102                f"{exit_points[i].name} -> {entry_points[i + 1].name}",
103            )
104            exit_points[i].add_transition(DirectTransition(entry_points[i + 1]))
105        return True
106
107    @staticmethod
108    def _resolve_for_entry(element: BaseTaskNode | BaseSubGraph) -> BaseTaskNode:
109        """Resolve a TaskNode or SubGraph to its entry point.
110
111        Args:
112            element (BaseTaskNode | BaseSubGraph): The element to resolve.
113
114        Returns:
115            BaseTaskNode: The entry point of the resolved element.
116        """
117        if isinstance(element, BaseSubGraph):
118            return element.get_entry()
119        return element
120
121    @staticmethod
122    def _resolve_for_exits(element: BaseTaskNode | BaseSubGraph) -> BaseTaskNode:
123        """Resolve a TaskNode or SubGraph to its exit points.
124
125        Args:
126            element (BaseTaskNode | BaseSubGraph): The element to resolve.
127
128        Returns:
129            BaseTaskNode: The exit point of the resolved element.
130        """
131        if isinstance(element, BaseSubGraph):
132            return element.get_exits()[0]
133        return element

class NavigationTask(strategy.core.tasks.base_navigation_task.BaseNavigationTask[boombot_strategy.winter_game_context.WinterGameContext]): 
15class NavigationTask(BaseNavigationTask[WinterGameContext]):
16    """Thin wrapper exposing navigation to the strategy graph."""
17
18    @override
19    def handle(self, ctx: WinterGameContext) -> bool:
20        """Execute the navigation plan and update the rolling basis.
21
22        Args:
23            ctx (WinterGameContext): The current game context.
24
25        Returns:
26            bool: ``True`` if the navigation task is finished, ``False`` otherwise.
27        """
28        if not self._is_initialized:
29            self._initialize(ctx)
30
31        ctx.arena.ally_zone.update(
32            ctx.arena.team_color,
33            ctx.rolling_basis.odometrie,
34            ctx.arena.enemy_zone.point,
35        )
36
37        cmd: TrajectoryPlanCommand = self.navigator_task.handle(
38            ally_zone=ctx.arena.ally_zone,
39            enemy_zone=ctx.arena.enemy_zone,
40        )
41
42        ctx.rolling_basis.set_target_position(cmd.get_position_command())
43        self._logger.debug(f"[TASK:Nav] {cmd.get_full_command()}")
44        self._logger.debug(f"[TASK:Nav] State: {self.navigator_task.state}")
45        return self.navigator_task.state.is_finished()

class SetOdometrie(strategy.core.tasks.base_task.BaseTask[boombot_strategy.winter_game_context.WinterGameContext]): 
13class SetOdometrie(BaseTask[WinterGameContext]):
14    """Task to update the robot's odometry position.
15
16    The task uses the provided ``x``, ``y`` and ``theta`` values if given;
17    otherwise it falls back to the current values from the rolling basis.
18    After updating the odometry, the arena's ally zone is synchronized.
19    """
20
21    def __init__(
22        self,
23        x: float | None = None,
24        y: float | None = None,
25        theta: float | None = None,
26    ) -> None:
27        """Initialize the SetOdometrie task with optional position values.
28
29        Args:
30            x (float | None, optional): X coordinate. Defaults to None.
31            y (float | None, optional): Y coordinate. Defaults to None.
32            theta (float | None, optional): Orientation in radians. Defaults to None.
33        """
34        self.x: float | None = x
35        self.y: float | None = y
36        self.theta: float | None = theta
37
38        super().__init__()
39
40    @override
41    def handle(self, ctx: WinterGameContext) -> bool:
42        """Handle the execution of the odometry setting task.
43
44        Args:
45            ctx (WinterGameContext): Context containing game state and arena info.
46
47        Returns:
48            bool: Always returns ``True`` after setting the new odometry.
49        """
50        # CRITICAL: Get current position from rolling_basis.odometrie
51        # (not arena.ally_zone.point) because arena.ally_zone.point may be outdated
52        current_position: OrientedPoint = ctx.rolling_basis.odometrie
53
54        # Create new position using provided values or current ones as fallback
55        new_position = OrientedPoint(
56            self.x if self.x is not None else current_position.x,
57            self.y if self.y is not None else current_position.y,
58            self.theta if self.theta is not None else current_position.theta,
59        )
60
61        # Update rolling basis odometry
62        ctx.rolling_basis.set_odometrie(new_position)
63
64        # Synchronize arena.ally_zone with the new odometry
65        ctx.arena.ally_zone.update(
66            ctx.arena.team_color,
67            new_position,
68            ctx.arena.enemy_zone.point,
69        )
70
71        return True

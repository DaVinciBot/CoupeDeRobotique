110    def rcv_rolling_basis_state(self, msg: bytes) -> None:
111        """Handle rolling basis odometry update messages from the Teensy.
112
113        The message contains:
114        - float x: X-coordinate of the position (4 bytes).
115        - float y: Y-coordinate of the position (4 bytes).
116        - float theta: Orientation (4 bytes).
117        - float current_linear_speed: Current linear speed (4 bytes).

119
120        Args:
121            msg (bytes): The received message bytes.
122        """
123        # Unpack new odometry
124        self.odometrie = OrientedPoint(
125            (struct.unpack("<d", msg[0:8])[0], struct.unpack("<d", msg[8:16])[0]),
126            struct.unpack("<d", msg[16:24])[0],
127        )
128        # After receiving actual state, log entry

338    def _load_pid(
339        *args: float | dict[str, float],
340        **kwargs: float,
341    ) -> PID:
342        """Load the PID values.
343
344        Overloads:
345            - set_linear_position_pid(float, float, float) → None
346            - set_linear_position_pid(dict[str, float]) → None
347            - set_linear_position_pid(kp=float, ki=float, kd=float) → None
348
349        Args:
350            *args (float | dict[str, float]): Either three floats (kp, ki, kd) or a
351                single dictionary with keys 'kp', 'ki', 'kd'.
352            **kwargs (float): Keyword arguments mapping PID fields to values.
353
354        Returns:
355            PID: The configured PID instance.
356
357        Raises:
358            ValueError: If the arguments do not match any expected format.
359        """
360        if len(args) == 3 and all(isinstance(arg, float) for arg in args):  # noqa: PLR2004
361            pid = PID(*args)  # pyright: ignore[reportArgumentType]
362        elif len(args) == 1 and isinstance(args[0], dict):
363            pid = PID.from_dict(args[0])
364        elif kwargs:
365            pid = PID.from_dict(kwargs)
366        else:
367            msg = "Invalid arguments for PID configuration."
368            raise ValueError(msg)
369        return pid

207    def _simulate_step_unlocked(self, dt: float) -> None:
208        """Integrate motion assuming the caller already holds the lock.
209
210        Args:
211            dt (float): Time in seconds to simulate the motion for.
212
213        Raises:
214            ValueError:
215                If odometrie.theta is None, which is required for motion simulation.
216        """
217        if dt <= 0.0:
218            return
219
220        if self.odometrie.theta is None:
221            msg = "Cannot simulate motion with undefined theta."
222            raise ValueError(msg)
223
224        delta_distance = self.linear_speed * dt
225        delta_theta = self.angular_speed * dt
226
227        heading_mid = self.odometrie.theta + (delta_theta / 2.0)
228        new_x = self.odometrie.x + math.cos(heading_mid) * delta_distance
229        new_y = self.odometrie.y + math.sin(heading_mid) * delta_distance
230        new_theta = self._normalize_angle(self.odometrie.theta + delta_theta)
231        left_wheel_delta_cm = delta_distance - (delta_theta * 15.45)
232        right_wheel_delta_cm = delta_distance + (delta_theta * 15.45)
233
234        self.odometrie = OrientedPoint((new_x, new_y), new_theta)
235        assert self.odometrie.theta is not None
236        self.left_wheel_position_cm += left_wheel_delta_cm
237        self.right_wheel_position_cm += right_wheel_delta_cm
238        self.left_wheel_target_cm = self.left_wheel_position_cm
239        self.right_wheel_target_cm = self.right_wheel_position_cm
240        if self._debug_recorder.enabled:
241            self._debug_recorder.set_odometry(
242                self.odometrie,
243                measured_linear_speed=self.linear_speed,
244                measured_angular_speed=self.angular_speed,
245            )
246            dx = self.target_position.x - self.odometrie.x
247            dy = self.target_position.y - self.odometrie.y
248            linear_error = (
249                math.cos(self.odometrie.theta) * dx
250                + math.sin(self.odometrie.theta) * dy
251            )
252            target_theta = self.target_position.theta
253            if target_theta is None:
254                target_theta = self.odometrie.theta
255            angular_error = self._normalize_angle(target_theta - self.odometrie.theta)
256            self._debug_recorder.set_pid_telemetry(
257                target_position=self.target_position,
258                linear_error=linear_error,
259                angular_error=angular_error,
260                linear_output=delta_distance,
261                angular_output=delta_theta,
262                left_wheel_target_cm=self.left_wheel_target_cm,
263                right_wheel_target_cm=self.right_wheel_target_cm,
264                left_wheel_position_cm=self.left_wheel_position_cm,
265                right_wheel_position_cm=self.right_wheel_position_cm,
266                left_wheel_error_cm=0.0,
267                right_wheel_error_cm=0.0,
268                left_pwm=0,
269                right_pwm=0,
270                left_ticks=0,
271                right_ticks=0,
272                left_delta_ticks=0,
273                right_delta_ticks=0,
274            )
